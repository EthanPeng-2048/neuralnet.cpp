// ── batched_matmul.comp ──────────────────────────────────────────────────
// 批量分块矩阵乘法原语：对每个 batch b 计算 C_b = alpha * op(A_b, B_b)
//
// 布局约定：
//   A: (batch * A_rows_per_batch, A_cols) 行主序 — 按 batch 切分为连续行块
//   B: (batch * B_rows_per_batch, B_cols) 行主序
//   C: (batch * M, N) 行主序 — 每个批次结果垂直堆叠
//
// 转置标志（与 matmul.comp 语义一致）：
//   transA=0: A_b 为 (M, K)，A_b[m][k] = A[b*M*K + m*K + k]
//   transA=1: A_b 存储为 (K, M) 按 A_b^T 使用，A_b[m][k] = A[b*M*K + k*M + m]
//   transB=0: B_b 为 (K, N)，B_b[k][n] = B[b*K*N + k*N + n]
//   transB=1: B_b 存储为 (N, K) 按 B_b^T 使用，B_b[k][n] = B[b*K*N + n*K + k]
//
// ── 与 matmul_tiled.comp 同配方（v3 移植，2026-09-24）─────────────────────
// 本 shader 长期停留在最初的标量 4×4 版（每 k 8 次标量 LDS + 每 tile 2 次
// barrier + 标量全局访问带逐元素 /,% 与边界分支），而 matmul_tiled 已演进到
// v3 —— 同形状下 batched 比 matmul 慢 1.85×（1024³ 实测 1.51 vs 0.81ms）。
// 移植的三重配方（逐项与旧版的差异）：
//   ① vec4 共享 Ash[k][m]/Bsh[k][n]：每线程每 k 仅 2 次 vec4 LDS + 16 FMA
//      （8:1 计算:加载比，旧版标量 2:1）
//   ② BK=16 双缓冲：tile t+1 的全局加载与 tile t 的计算并行发射，每 tile
//      barrier 2→1 次，全局延迟（~600 周期）被计算覆盖
//   ③ vec4 全局快路径：按转置形态选连续维分组，对齐时整组一条 vec4 访问
//      （旧版每元素 1 次标量 ldg/stg + /,% + 边界判断）
//
// vec4 对齐条件（含 batch 基址 b*step：step%4==0 由下列条件蕴含）：
//   A: transA=0 需 K%4==0（step=M*K ⇒ K%4⇒step%4）；transA=1 需 M%4==0
//   B: transB=0 需 N%4==0；transB=1 需 K%4==0（step=K*N）
//   C: 写出需 N%4==0（step=M*N）
//   基址 16B 对齐由 MemoryPool 分配对齐 ≥16 保证（compute_memory_pool.hpp）。
//   未对齐（奇数形状）或线程持边界组：回退同组标量 + 边界判零，覆盖与
//   vec4 路径完全一致 → 每个输出元素恰被写一次，语义不变。
//
// 优化策略：
//   - WorkGroup：16×16 线程，Z 维度索引 batch，计算 64×64 输出块
//   - alpha 在写出时一次完成（cuBLAS sgemm 语义，供上层折叠 1/sqrt(d_k)
//     等系数，省去额外的全矩阵 scale pass）
//   - Dispatch: (ceil(N/64), ceil(M/64), batch)；PC: M,N,K,transA,transB,alpha
//
// 注意：此着色器只包含原语操作（矩阵乘法的批量索引 + 标量缩放），
// 不含任何算法语义。
// ─────────────────────────────────────────────────────────────────────────

#version 450

// ── 精度变体（编译期开关：glslc -DNN_SHADER_F16=1）────────────────────────
// 默认（无宏）= 全 f32，**下方 f32 分支文本逐字未改**（零回归）。
// 带宏 = f16 存储变体：缓冲声明 float16_t（GL_EXT_shader_16bit_storage），
//   别名槽改 uvec2（4×half = 8B，std430 步长恰好 8）→ unpackHalf2x16 解出
//   vec4；全局读处统一 float(...) 转 f32、写出处统一 float16_t(...)。
//   **共享 tile / vec4 外积累加 / 分块 / 双缓冲 / barrier 节奏与 f32 完全同构**
//   ——只有全局加载/写出的元素类型变了，符合 §7.2「f32 参考累加 + 输出舍入」。
//   C 不提供 vec4 别名（f16 走逐标量写出，与 N%4!=0 的回退路径同构）。
#if defined(NN_SHADER_F16)
#extension GL_EXT_shader_16bit_storage : require
#define NN_ETYPE float16_t
#define NN_V4    uvec2
#define NN_LOADV4(arr, i) vec4(unpackHalf2x16(arr[i].x), unpackHalf2x16(arr[i].y))
#define NN_RD(x) float(x)
#define NN_WR(x) float16_t(x)
#else
#define NN_ETYPE float
#define NN_V4    vec4
#define NN_LOADV4(arr, i) arr[i]
#define NN_RD(x) x
#define NN_WR(x) x
#endif

layout(local_size_x = 16, local_size_y = 16) in;

layout(std430, binding = 0) readonly buffer  BufferA { NN_ETYPE A[]; };
layout(std430, binding = 1) readonly buffer  BufferB { NN_ETYPE B[]; };
layout(std430, binding = 2) writeonly buffer BufferC { NN_ETYPE C[]; };

// vec4 别名视图（同 binding 双声明；仅在元素下标 %4==0 的快路径使用）
layout(std430, binding = 0) readonly buffer  BufferAv4 { NN_V4 Av4[]; };
layout(std430, binding = 1) readonly buffer  BufferBv4 { NN_V4 Bv4[]; };
#if !defined(NN_SHADER_F16)
layout(std430, binding = 2) writeonly buffer BufferCv4 { vec4 Cv4[]; };
#endif

// Push constants：单 batch 逻辑维度 + 转置标志 + 输出缩放（batch 步长可推导）
layout(push_constant) uniform PushConstants {
    uint M;      // 每个 batch 的输出行数
    uint N;      // 每个 batch 的输出列数
    uint K;      // 收缩维度
    uint transA; // 1 = A_b 存储为 (K,M)，按 A_b^T 使用
    uint transB; // 1 = B_b 存储为 (N,K)，按 B_b^T 使用
    float alpha; // 输出缩放系数：C_b = alpha * op(A_b, B_b)
};

// 分块维度常量（与 matmul_tiled 保持同步；2026-09-24 OP/融合统一 A/B：
// OP 级 BK=32 四点全回退 → 定案 BK=16 双缓冲 16KB）
const uint BM = 64u;
const uint BN = 64u;
const uint BK = 16u;

// 每线程协作加载元素数（BM*BK/256 → 4；BK=16 时 groups=1，改 BK 前先验 %4==0）
const uint EPT_A = (BM * BK) / 256u;
const uint EPT_B = (BK * BN) / 256u;

// Shared Memory（vec4 布局，16B 对齐，双缓冲）：
//   Ash[stage][k][m]：A^T 分块（转置写共享内存，内层按行连续读）
//   Bsh[stage][k][n]：B 分块
shared vec4 Ash[2][BK][BM / 4];
shared vec4 Bsh[2][BK][BN / 4];

// ── 协作加载 A 分块 BM×BK → 指定 stage（base = b*M*K）────────────────────
void load_A_tile(uint t, uint stage, uint tid, uint block_row, uint base)
{
    const uint a_base_col = t * BK;
    const bool k_contig = (transA == 0u);
    // 沿 k 连续需 K%4==0；沿 m 连续需 M%4==0（两者均蕴含 base%4==0）
    const bool aligned = k_contig ? ((K & 3u) == 0u) : ((M & 3u) == 0u);
    const uint groups = EPT_A / 4u;

    for (uint vv = 0u; vv < groups; ++vv)
    {
        const uint g = tid * groups + vv;
        if (k_contig)
        {
            // 沿 k 连续：4 个连续 k，同一行 m（base + gr*K + gc，gc%4==0）
            const uint m  = (4u * g) / BK;
            const uint k0 = (4u * g) % BK;
            const uint gr = block_row + m;
            const uint gc = a_base_col + k0;
            if (aligned && gr < M && gc < K)
            {
                const vec4 v = NN_LOADV4(Av4, (base + gr * K + gc) >> 2);
                Ash[stage][k0    ][m >> 2][m & 3u] = v.x;
                Ash[stage][k0 + 1u][m >> 2][m & 3u] = v.y;
                Ash[stage][k0 + 2u][m >> 2][m & 3u] = v.z;
                Ash[stage][k0 + 3u][m >> 2][m & 3u] = v.w;
            }
            else
            {
                for (uint j = 0u; j < 4u; ++j)
                {
                    const float v = (gr < M && gc + j < K) ? NN_RD(A[base + gr * K + gc + j]) : 0.0;
                    Ash[stage][k0 + j][m >> 2][m & 3u] = v;
                }
            }
        }
        else
        {
            // 沿 m 连续（A 存储 K×M）：4 个连续 m，同一 k（base + gc*M + gr）
            const uint mq = g % (BM / 4u);
            const uint k  = g / (BM / 4u);
            const uint gc = a_base_col + k;
            const uint gr = block_row + mq * 4u;
            if (aligned && gc < K && gr < M)
            {
                Ash[stage][k][mq] = NN_LOADV4(Av4, (base + gc * M + gr) >> 2);
            }
            else
            {
                for (uint j = 0u; j < 4u; ++j)
                {
                    const float v = (gr + j < M && gc < K) ? NN_RD(A[base + gc * M + gr + j]) : 0.0;
                    Ash[stage][k][mq][j] = v;
                }
            }
        }
    }
}

// ── 协作加载 B 分块 BK×BN → 指定 stage（base = b*K*N）────────────────────
void load_B_tile(uint t, uint stage, uint tid, uint block_col, uint base)
{
    const uint b_base_row = t * BK;
    const bool n_contig = (transB == 0u);
    // 沿 n 连续需 N%4==0；沿 k 连续需 K%4==0（两者均蕴含 base%4==0）
    const bool aligned = n_contig ? ((N & 3u) == 0u) : ((K & 3u) == 0u);
    const uint groups = EPT_B / 4u;

    for (uint vv = 0u; vv < groups; ++vv)
    {
        const uint g = tid * groups + vv;
        if (n_contig)
        {
            // 沿 n 连续：4 个连续 n，同一行 k（base + gr*N + gc）
            const uint k  = (4u * g) / BN;
            const uint n0 = (4u * g) % BN;
            const uint gr = b_base_row + k;
            const uint gc = block_col + n0;
            if (aligned && gr < K && gc < N)
            {
                const vec4 v = NN_LOADV4(Bv4, (base + gr * N + gc) >> 2);
                Bsh[stage][k][n0 >> 2] = v;
            }
            else
            {
                for (uint j = 0u; j < 4u; ++j)
                {
                    const float v = (gr < K && gc + j < N) ? NN_RD(B[base + gr * N + gc + j]) : 0.0;
                    Bsh[stage][k][(n0 + j) >> 2][(n0 + j) & 3u] = v;
                }
            }
        }
        else
        {
            // 沿 k 连续（B 存储 N×K）：4 个连续 k，同一 n（base + gc*K + gr）
            const uint n  = g / 4u;
            const uint k0 = (g % 4u) * 4u;
            const uint gc = block_col + n;
            const uint gr = b_base_row + k0;
            if (aligned && gc < N && gr < K)
            {
                const vec4 v = NN_LOADV4(Bv4, (base + gc * K + gr) >> 2);
                for (uint j = 0u; j < 4u; ++j)
                    Bsh[stage][k0 + j][n >> 2][n & 3u] = v[j];
            }
            else
            {
                for (uint j = 0u; j < 4u; ++j)
                {
                    const float v = (gr + j < K && gc < N) ? NN_RD(B[base + gc * K + gr + j]) : 0.0;
                    Bsh[stage][k0 + j][n >> 2][n & 3u] = v;
                }
            }
        }
    }
}

void main()
{
    const uint tx = gl_LocalInvocationID.x;   // 0..15
    const uint ty = gl_LocalInvocationID.y;   // 0..15
    const uint tid = ty * 16u + tx;           // 线性线程 ID 0..255
    const uint b = gl_WorkGroupID.z;          // batch 索引

    // 当前 WorkGroup 负责的 batch 内输出块起始坐标 + 各 buffer 的 batch 基址
    const uint block_row = gl_WorkGroupID.y * BM;
    const uint block_col = gl_WorkGroupID.x * BN;
    const uint base_a = b * M * K;
    const uint base_b = b * K * N;
    const uint base_c = b * M * N;

    // ── 寄存器分块：4×4 累加器（vec4） ─────────────────────────────
    vec4 acc[4];
    for (uint i = 0u; i < 4u; ++i)
        acc[i] = vec4(0.0);

    const uint num_tiles = (K + BK - 1u) / BK;

    // ── 前奏：加载 tile 0 → stage 0，同步一次 ───────────────────────
    load_A_tile(0u, 0u, tid, block_row, base_a);
    load_B_tile(0u, 0u, tid, block_col, base_b);
    barrier();

    // ── 流水循环：发射 tile t+1 的加载（stage^1），同时计算 tile t ────
    uint stage = 0u;
    for (uint t = 0u; t < num_tiles; ++t)
    {
        const uint nxt = t + 1u;
        const uint nstage = stage ^ 1u;
        if (nxt < num_tiles)
        {
            load_A_tile(nxt, nstage, tid, block_row, base_a);
            load_B_tile(nxt, nstage, tid, block_col, base_b);
        }

        // ── vec4 外积累加（每 k：2 次 vec4 LDS + 16 FMA = 8:1） ──────
        #pragma unroll 4
        for (uint k = 0u; k < BK; ++k)
        {
            const vec4 a_vec = Ash[stage][k][ty];
            const vec4 b_vec = Bsh[stage][k][tx];

            acc[0] += a_vec.x * b_vec;
            acc[1] += a_vec.y * b_vec;
            acc[2] += a_vec.z * b_vec;
            acc[3] += a_vec.w * b_vec;
        }

        barrier();
        stage = nstage;
    }

    // ── 写入结果（batch 偏移 = b * M * N，含 alpha 缩放）────────────
#if defined(NN_SHADER_F16)
    // f16 输出：逐标量写出（C 无 vec4 别名视图），与 f32 的 N%4!=0 回退同构
    for (uint i = 0u; i < 4u; ++i)
    {
        const uint out_row = block_row + ty * 4u + i;
        if (out_row >= M) continue;
        const uint out_col0 = block_col + tx * 4u;
        if (out_col0 >= N) continue;
        C[base_c + out_row * N + out_col0 + 0u] = NN_WR(alpha * acc[i].x);
        if (out_col0 + 1u < N)
            C[base_c + out_row * N + out_col0 + 1u] = NN_WR(alpha * acc[i].y);
        if (out_col0 + 2u < N)
            C[base_c + out_row * N + out_col0 + 2u] = NN_WR(alpha * acc[i].z);
        if (out_col0 + 3u < N)
            C[base_c + out_row * N + out_col0 + 3u] = NN_WR(alpha * acc[i].w);
    }
#else
    const bool c_vec4 = ((N & 3u) == 0u);
    for (uint i = 0u; i < 4u; ++i)
    {
        const uint out_row = block_row + ty * 4u + i;
        if (out_row >= M) continue;
        const uint out_col0 = block_col + tx * 4u;
        if (c_vec4)
        {
            if (out_col0 < N)
                Cv4[(base_c + out_row * N + out_col0) >> 2] = alpha * acc[i];
        }
        else
        {
            if (out_col0 >= N) continue;
            C[base_c + out_row * N + out_col0 + 0u] = alpha * acc[i].x;
            if (out_col0 + 1u < N)
                C[base_c + out_row * N + out_col0 + 1u] = alpha * acc[i].y;
            if (out_col0 + 2u < N)
                C[base_c + out_row * N + out_col0 + 2u] = alpha * acc[i].z;
            if (out_col0 + 3u < N)
                C[base_c + out_row * N + out_col0 + 3u] = alpha * acc[i].w;
        }
    }
#endif
}

// ── cast.comp — 通用精度转换原语 ───────────────────────────────────────
// 目标：engine.cast(tensor, precision) 的 GPU 原生实现（无 PCIe 往返）。
// 源/目标均以原始 32-bit word 缓冲读写（float=4B，f16 按 2B 紧密打包）：
//   - f16→f32（kind=0）：S[w] = 两个 half 打包成 uint；D[2w/2w+1] = float words
//   - f32→f16（kind=1，round-half-to-even）：S[2w/2w+1] = float words；
//              D[w] = unpackHalf2x16 逆操作 packHalf2x16 打包两个 half
// 同精度 cast 是别名不落 kernel；BF16/F64 为保留值，后端分派处清晰报错。
// 后续精度对（BF16/F64/f8 等）扩 kind 枚举 + 后端 (src,dst) 分派表即可，
// 引擎 API 恒为 cast(tensor, precision)，不因精度增加而改变。
// ───────────────────────────────────────────────────────────────────────
#version 450
layout(local_size_x = 256) in;

layout(std430, binding = 0) readonly  buffer BufS { uint S[]; };
layout(std430, binding = 1) writeonly buffer BufD { uint D[]; };

layout(push_constant) uniform Push {
    uint count;   // 元素数 rows*cols（源与目标元素数相同，仅字节宽度差）
    uint kind;    // 0 = f16→f32，1 = f32→f16
};

void main() {
    const uint pairs = (count + 1u) >> 1u;   // 每线程处理 2 元素（一对 half 或单词对）
    const uint gsize = gl_NumWorkGroups.x * gl_NumWorkGroups.y
                     * gl_NumWorkGroups.z * 256u;

    if (kind == 0u)
    {
        // f16 → f32（升精度，精确无损）
        for (uint w = gl_GlobalInvocationID.x; w < pairs; w += gsize)
        {
            const vec2 h = unpackHalf2x16(S[w]);
            D[2u * w] = floatBitsToUint(h.x);
            if (2u * w + 1u < count)
                D[2u * w + 1u] = floatBitsToUint(h.y);
        }
    }
    else
    {
        // f32 → f16（降精度，packHalf2x16 按 round-half-to-even）
        for (uint w = gl_GlobalInvocationID.x; w < pairs; w += gsize)
        {
            const uint i0 = 2u * w;
            const float x = uintBitsToFloat(S[i0]);
            const float y = (i0 + 1u < count) ? uintBitsToFloat(S[i0 + 1u]) : 0.0f;
            D[w] = packHalf2x16(vec2(x, y));
        }
    }
}
// ── col2im.comp ───────────────────────────────────────────────────────────
// 纯原语：im2col 的伴随（adjoint / 反向散射），数据搬运，绝不含算法
//
//   col (C*k*k, B*OH*OW) → out (C*H*W, B)
//   out[ci*H*W + ih*W + iw, b]
//     = Σ_{kh,kw} col[(ci*k + kh)*k + kw, (oh*OW + ow)*B + b]
//   其中 oh = (ih + pad - kh)/stride（须整除且落在 [0, OH)），ow 同理。
//   **重叠窗口（stride < k）的贡献在此累加**；每个输出元素由单个线程完整
//   求和，故不需要原子操作、也不需要预先清零输出。
//
// 绑定布局：
//   binding 0: 输入 col (readonly,  C*k*k × B*OH*OW)
//   binding 1: 输出 out (writeonly, C*H*W × B)
//
// Push Constants（36 bytes）：
//   uint C, H, W, k, stride, pad, OH, OW, B
// ─────────────────────────────────────────────────────────────────────────

#version 450

layout(local_size_x = 256) in;

layout(std430, binding = 0) readonly  buffer BufIn  { float in_data[];  };
layout(std430, binding = 1) writeonly buffer BufOut { float out_data[]; };

layout(push_constant) uniform PushConstants {
    uint C;
    uint H;
    uint W;
    uint k;
    uint stride;
    uint pad;
    uint OH;
    uint OW;
    uint B;
};

void main()
{
    const uint idx = gl_GlobalInvocationID.x;
    const uint rows = C * H * W;
    const uint cols = B;
    if (idx >= rows * cols) return;

    const uint row = idx / cols;   // 输入像素行 = ci*H*W + ih*W + iw
    const uint b   = idx % cols;

    const uint ci  = row / (H * W);
    const uint rem = row % (H * W);
    const uint ih  = rem / W;
    const uint iw  = rem % W;

    const uint P = OH * OW;
    const int s_stride = int(stride);

    float acc = 0.0;
    for (uint kh = 0; kh < k; ++kh)
    {
        const int t = int(ih + pad) - int(kh);
        if (t < 0 || (t % s_stride) != 0) continue;
        const int oh = t / s_stride;
        if (oh >= int(OH)) continue;
        for (uint kw = 0; kw < k; ++kw)
        {
            const int s = int(iw + pad) - int(kw);
            if (s < 0 || (s % s_stride) != 0) continue;
            const int ow = s / s_stride;
            if (ow >= int(OW)) continue;
            const uint r = (ci * k + kh) * k + kw;
            acc += in_data[r * (B * P) + (uint(oh) * OW + uint(ow)) * B + b];
        }
    }
    out_data[idx] = acc;
}

// ── elementwise_v2.comp ───────────────────────────────────────────────────
// 纯原语逐元素运算着色器（不含任何算法）
//
// 设计原则：本着色器只提供 op-level 原语（加减乘除、exp、sqrt、max、min、
// select 等），绝不包含 ReLU、GeLU、Softmax、LayerNorm 等算法。
// 算法由 Layer 层通过组合原语表达。
//
// 绑定布局：
//   binding 0: 输入 A（条件操作数 1 / 主操作数）
//   binding 1: 输入 B（条件操作数 2 / then 张量 / 次操作数）
//   binding 2: 输入 C（else 张量，仅 SELECT 模式使用）
//   binding 3: 输出 OUT
//
// Push Constants：
//   uint  count       — 元素总数
//   uint  mode        — 运算模式（见下）
//   uint  op          — 一元/二元算子编号（mode 0/1 使用）
//   uint  cmp_op      — 比较算子编号（mode 2 SELECT 使用）
//   uint  flags       — 位标志：
//                         bit0: B 为标量（用 scalar_b 替代 B[idx]）
//                         bit1: then 为标量（用 scalar_then 替代 B[idx]）
//                         bit2: else 为标量（用 scalar_else 替代 C[idx]）
//                         bit3: 二元运算标量在前（scalar op A，仅 mode 1）
//   float scalar_b    — 标量 B / 条件操作数 2 标量
//   float scalar_then — then 标量值
//   float scalar_else — else 标量值
//
// 运算模式（mode）：
//   0 = UNARY:    out = unary_op(A[idx])
//   1 = BINARY:   out = binary_op(a, b)
//                    a = A[idx]
//                    b = (flags & 1) ? scalar_b : B[idx]
//                    若 flags & 8: out = binary_op(b, a)  [标量在前]
//   2 = SELECT:   out = compare_op(a, b) ? then_val : else_val
//                    a = A[idx]
//                    b = (flags & 1) ? scalar_b : B[idx]
//                    then_val = (flags & 2) ? scalar_then : B[idx]
//                    else_val = (flags & 4) ? scalar_else : C[idx]
//   3 = AXPY:     out = A[idx] + scalar_b * B[idx]   (融合 axpy：1 次 dispatch 替代 3 次)
//                  用于 Optimizer 的 scale_add_(dst, scalar, src) 操作
//
// 一元算子（op）：
//   0=Neg, 1=Exp, 2=Log, 3=Sqrt, 4=Rsqrt, 5=Abs, 6=Tanh
//
// 二元算子（op）：
//   0=Add, 1=Sub, 2=Mul, 3=Div, 4=Max, 5=Min
//
// 比较算子（cmp_op）：
//   0=Lt, 1=Le, 2=Gt, 3=Ge, 4=Eq, 5=Ne
//
// ── vec4 kernel（2026-09-24，与 DSL 融合逐元素 vec4 路径同步）──────────────
// DSL 侧（expr_glsl_gen glsl_vec4_eligible + vec4 kernel）早已按「每线程 4 个
// 相邻元素」发射 vec4 加载/运算/存储，dispatch 宽度 4；OP 级此前是标量
// 每线程 1 元素。本 shader 现镜像同一契约：
//   - main：base = gid*4，base >= count 返回；
//   - 快路径（count%4==0 且 base+3 < count）→ 全 vec4（索引 flat、4 相邻
//     元素连续 ⇒ vec4 别名视图整读整写；16B 对齐由 MemoryPool 保证）；
//   - 回退：同 kernel 内 4 次标量迭代（count%4≠0 的尾组/奇数 count），
//     快路径与回退覆盖完全一致 → 每个元素恰被处理一次；
//   - backend dispatch：ceil(count / (256*4))（与 run_fused_gpu 的
//     count/(256*vec_width) 同口径）。
// vec4 运算用 GLSL 内建重载（exp(vec4)、mix(vec4,vec4,bvec4) 等），
// 语义与标量路径逐位同构（exp 等超越数按 lane 独立求值）。
// ─────────────────────────────────────────────────────────────────────────

#version 450

layout(local_size_x = 256) in;

layout(std430, binding = 0) readonly buffer  BufA   { float buf_a[]; };
layout(std430, binding = 1) readonly buffer  BufB   { float buf_b[]; };
layout(std430, binding = 2) readonly buffer  BufC   { float buf_c[]; };
layout(std430, binding = 3) writeonly buffer BufOut { float buf_out[]; };

// vec4 别名视图（同 binding 双声明，std430；仅 count%4==0 快路径使用）
layout(std430, binding = 0) readonly buffer  BufAv4 { vec4 buf_a4[]; };
layout(std430, binding = 1) readonly buffer  BufBv4 { vec4 buf_b4[]; };
layout(std430, binding = 2) readonly buffer  BufCv4 { vec4 buf_c4[]; };
layout(std430, binding = 3) writeonly buffer BufOut4 { vec4 buf_out4[]; };

layout(push_constant) uniform PushConstants {
    uint  count;
    uint  mode;
    uint  op;
    uint  cmp_op;
    uint  flags;
    float scalar_b;
    float scalar_then;
    float scalar_else;
};

// ── 一元运算（标量 / vec4 同名重载，语义逐 lane 同构）─────────────────────
float apply_unary(uint op, float x)
{
    if (op == 0u) return -x;                          // Neg
    if (op == 1u) return exp(x);                      // Exp
    if (op == 2u) return log(x);                      // Log
    if (op == 3u) return sqrt(x);                     // Sqrt
    if (op == 4u) return 1.0 / sqrt(x);               // Rsqrt
    if (op == 5u) return abs(x);                      // Abs
    if (op == 6u) return tanh(x);                     // Tanh
    return 0.0;
}
vec4 apply_unary(uint op, vec4 x)
{
    if (op == 0u) return -x;
    if (op == 1u) return exp(x);
    if (op == 2u) return log(x);
    if (op == 3u) return sqrt(x);
    if (op == 4u) return 1.0 / sqrt(x);
    if (op == 5u) return abs(x);
    if (op == 6u) return tanh(x);
    return vec4(0.0);
}

// ── 二元运算 ──────────────────────────────────────────────────────────────
float apply_binary(uint op, float a, float b)
{
    if (op == 0u) return a + b;                       // Add
    if (op == 1u) return a - b;                       // Sub
    if (op == 2u) return a * b;                       // Mul
    if (op == 3u) return a / b;                       // Div
    if (op == 4u) return max(a, b);                   // Max
    if (op == 5u) return min(a, b);                   // Min
    return 0.0;
}
vec4 apply_binary(uint op, vec4 a, vec4 b)
{
    if (op == 0u) return a + b;
    if (op == 1u) return a - b;
    if (op == 2u) return a * b;
    if (op == 3u) return a / b;
    if (op == 4u) return max(a, b);
    if (op == 5u) return min(a, b);
    return vec4(0.0);
}

// ── 比较运算（vec4 返回 bvec4，供 mix 按 lane 选择）────────────────────────
bool apply_compare(uint op, float a, float b)
{
    if (op == 0u) return a <  b;                      // Lt
    if (op == 1u) return a <= b;                      // Le
    if (op == 2u) return a >  b;                      // Gt
    if (op == 3u) return a >= b;                      // Ge
    if (op == 4u) return a == b;                      // Eq
    if (op == 5u) return a != b;                      // Ne
    return false;
}
bvec4 apply_compare(uint op, vec4 a, vec4 b)
{
    if (op == 0u) return lessThan(a, b);
    if (op == 1u) return lessThanEqual(a, b);
    if (op == 2u) return greaterThan(a, b);
    if (op == 3u) return greaterThanEqual(a, b);
    if (op == 4u) return equal(a, b);
    if (op == 5u) return notEqual(a, b);
    return bvec4(false);
}

// ── 标量主路径（回退：每元素一次；idx 由调用方给出）───────────────────────
float eval_scalar(uint idx)
{
    const float a = buf_a[idx];

    // ── 模式 0：一元运算 ──
    if (mode == 0u)
        return apply_unary(op, a);

    // ── 模式 1：二元运算 ──
    if (mode == 1u)
    {
        const float b = ((flags & 1u) != 0u) ? scalar_b : buf_b[idx];
        if ((flags & 8u) != 0u)
            return apply_binary(op, b, a);            // 标量在前
        else
            return apply_binary(op, a, b);            // 张量在前
    }

    // ── 模式 2：条件选择 ──
    if (mode == 2u)
    {
        const float b      = ((flags & 1u) != 0u) ? scalar_b      : buf_b[idx];
        const float then_v = ((flags & 2u) != 0u) ? scalar_then   : buf_b[idx];
        const float else_v = ((flags & 4u) != 0u) ? scalar_else   : buf_c[idx];
        return apply_compare(cmp_op, a, b) ? then_v : else_v;
    }

    // ── 模式 3：AXPY ──
    if (mode == 3u)
        return a + scalar_b * buf_b[idx];

    return 0.0;
}

// ── vec4 快路径（count%4==0 的整组 4 元素）────────────────────────────────
vec4 eval_vec4(uint base)
{
    const vec4 a = buf_a4[base >> 2];

    if (mode == 0u)
        return apply_unary(op, a);

    if (mode == 1u)
    {
        const vec4 b = ((flags & 1u) != 0u) ? vec4(scalar_b) : buf_b4[base >> 2];
        if ((flags & 8u) != 0u)
            return apply_binary(op, b, a);
        else
            return apply_binary(op, a, b);
    }

    if (mode == 2u)
    {
        const vec4 b      = ((flags & 1u) != 0u) ? vec4(scalar_b)    : buf_b4[base >> 2];
        const vec4 then_v = ((flags & 2u) != 0u) ? vec4(scalar_then) : buf_b4[base >> 2];
        const vec4 else_v = ((flags & 4u) != 0u) ? vec4(scalar_else) : buf_c4[base >> 2];
        // mix(else, then, cond)：cond = compare(a, b)（bvec4 按 lane 选择）
        return mix(else_v, then_v, apply_compare(cmp_op, a, b));
    }

    if (mode == 3u)
        return a + scalar_b * buf_b4[base >> 2];

    return vec4(0.0);
}

void main()
{
    const uint i = gl_GlobalInvocationID.x;
    const uint base = i * 4u;
    if (base >= count) return;

    // 快路径：整组 4 元素齐全且 count 对齐 → vec4 整读整写
    if ((count & 3u) == 0u && base + 3u < count)
    {
        buf_out4[base >> 2] = eval_vec4(base);
        return;
    }

    // 回退：同组 4 次标量迭代（尾组 / 奇数 count），覆盖与快路径一致
    for (uint j = 0u; j < 4u; ++j)
    {
        const uint e = base + j;
        if (e >= count) break;
        buf_out[e] = eval_scalar(e);
    }
}

// ── gather.comp ───────────────────────────────────────────────────────────
// 纯原语：按行索引从表中查表（gather_rows）
//
// 语义：
//   table: (vocab, D) — 行主序存储
//   indices: (num,)   — 行索引（float 编码的整数）
//   output: (num, D) — output[i] = table[indices[i]]
//
// 越界索引返回零行（防御性，与 CPU fallback 行为一致）。
//
// 绑定布局：
//   binding 0: table (readonly)
//   binding 1: indices (readonly)
//   binding 2: output (writeonly)
//
// Push Constants（12 bytes）：
//   uint vocab — table 行数
//   uint D     — 列宽（每行元素数）
//   uint num   — indices 元素个数
//
// ── vec4 + 按组求商（2026-09-24，与 DSL 侧「索引按组派生、免逐元素除模」
//    及 vec4 kernel 契约同步）────────────────────────────────────────────────
// 旧版每线程 1 元素、每元素一次 i=idx/D、d=idx%D 两笔整数除模 + 4 次标量
// 表读。现：
//   - 快路径（D%4==0 且 base+3<total）：base = gid*4 → i/d0 每组一次除模
//     （D%4 ⇒ d0%4 且 row*D 对齐 16B）→ indices[i] 一次、table vec4 整读、
//     out vec4 整写；越界行整组置零（同组同 i ⇒ 与逐元素置零语义一致）。
//   - 回退：同组 4 次标量迭代（奇数 D / 尾组），覆盖完全一致。
//   - dispatch：ceil(total/(256*4))（backend 侧同口径）。
// ─────────────────────────────────────────────────────────────────────────

#version 450

layout(local_size_x = 256) in;

layout(std430, binding = 0) readonly buffer  BufTable   { float table[];   };
layout(std430, binding = 1) readonly buffer  BufIndices { float indices[]; };
layout(std430, binding = 2) writeonly buffer BufOut     { float out_data[]; };

// vec4 别名视图（同 binding 双声明，std430；仅 D%4==0 快路径使用）
layout(std430, binding = 0) readonly buffer  BufTable4 { vec4 table4[];  };
layout(std430, binding = 2) writeonly buffer BufOut4   { vec4 out4[];    };

layout(push_constant) uniform PushConstants {
    uint vocab;
    uint D;
    uint num;
};

void main()
{
    const uint i = gl_GlobalInvocationID.x;
    const uint base = i * 4u;
    const uint total = num * D;
    if (base >= total) return;

    // ── 快路径：D%4==0 ⇒ 组内 4 元素同行、d0 对齐 16B ──
    if ((D & 3u) == 0u && base + 3u < total)
    {
        const uint ii  = base / D;                   // 索引数组下标（每组一次除法）
        const uint d0  = base - ii * D;
        const uint row = uint(indices[ii]);          // ← 表行号来自 indices[ii]
        if (row < vocab)
            out4[base >> 2] = table4[(row * D + d0) >> 2];
        else
            out4[base >> 2] = vec4(0.0);             // 越界行整组置零
        return;
    }

    // ── 回退：标量（每元素求 i/d；与旧版逐位同构）──
    for (uint j = 0u; j < 4u; ++j)
    {
        const uint e = base + j;
        if (e >= total) break;
        const uint ii = e / D;
        const uint d  = e % D;
        const uint row_idx = uint(indices[ii]);
        if (row_idx < vocab)
            out_data[e] = table[row_idx * D + d];
        else
            out_data[e] = 0.0;
    }
}

// ── group_reduce.comp ─────────────────────────────────────────────────────
// 分组归约原语（segmented reduction，沿行方向按固定长度 R 分组）
//
//   x (G*R, N) → out (G, N)
//   reduce_op=0 (sum): out[g][n] = Σ_{i<R} x[g*R + i][n]
//   reduce_op=1 (max): out[g][n] = max_i  x[g*R + i][n]
//
// 与 reduce.comp 的区别：归约轴不是整行/整列，而是每连续 R 行一组。
// 用途：池化窗口归约、多头分组统计等 —— 把"逐通道一次 dispatch"压成单次。
//
// 绑定布局：
//   binding 0: 输入 x   (readonly,  G*R × N)
//   binding 1: 输出 out (writeonly, G × N)
//
// Push Constants（16 bytes）：
//   uint G, uint R, uint N, uint reduce_op
//
// Dispatch：(ceil(G*N/256), 1, 1) —— 每线程一个输出元素，沿 R 串行累加；
//   线程在组内相邻（n 连续）→ 同一行的读地址连续、可合并访问。
//   （R 很大时若需要可再改工作组并行版；当前用途 R = 池化窗口/头数，较小。）
// ─────────────────────────────────────────────────────────────────────────

#version 450

layout(local_size_x = 256) in;

layout(std430, binding = 0) readonly  buffer BufIn  { float in_data[];  };
layout(std430, binding = 1) writeonly buffer BufOut { float out_data[]; };

layout(push_constant) uniform PushConstants {
    uint G;
    uint R;
    uint N;
    uint reduce_op;
};

void main()
{
    const uint idx = gl_GlobalInvocationID.x;
    if (idx >= G * N) return;

    const uint g = idx / N;
    const uint n = idx % N;
    const bool is_max = (reduce_op == 1u);

    // max 恒等元 = numeric_limits::lowest()（0xFF7FFFFF，2026-09-24 与
    // reduce.comp / DSL / CPU 全库统一；旧 -inf 在全 -inf 输入下与减法
    // 组合会出 −inf−−inf=NaN）
    float acc = is_max ? uintBitsToFloat(0xFF7FFFFFu) : 0.0;
    const uint base = g * R * N + n;
    for (uint i = 0u; i < R; ++i)
    {
        const float v = in_data[base + i * N];
        acc = is_max ? max(acc, v) : (acc + v);
    }
    out_data[idx] = acc;
}

// ── im2col.comp ───────────────────────────────────────────────────────────
// 纯原语：滑动窗展开（数据搬运，绝不含卷积/池化算法）
//
//   x (C*H*W, B) → out (C*k*k, B*OH*OW)
//   out[(ci*k + kh)*k + kw, (oh*OW + ow)*B + b]
//     = x[ci*H*W + (oh*stride + kh - pad)*W + (ow*stride + kw - pad), b]
//   越界（padding 区）取 0（列序为「位置优先」(oh, ow, b)）
//
// 绑定布局：
//   binding 0: 输入 x   (readonly,  C*H*W × B)
//   binding 1: 输出 out (writeonly, C*k*k × B*OH*OW)
//
// Push Constants（36 bytes）：
//   uint C, H, W, k, stride, pad, OH, OW, B
// ─────────────────────────────────────────────────────────────────────────

#version 450

layout(local_size_x = 256) in;

layout(std430, binding = 0) readonly  buffer BufIn  { float in_data[];  };
layout(std430, binding = 1) writeonly buffer BufOut { float out_data[]; };

layout(push_constant) uniform PushConstants {
    uint C;
    uint H;
    uint W;
    uint k;
    uint stride;
    uint pad;
    uint OH;
    uint OW;
    uint B;
};

void main()
{
    const uint idx = gl_GlobalInvocationID.x;
    const uint kk   = k * k;
    const uint rows = C * kk;
    const uint P    = OH * OW;
    const uint cols = B * P;
    if (idx >= rows * cols) return;

    const uint r = idx / cols;
    const uint c = idx % cols;

    const uint ci = r / kk;
    const uint kh = (r / k) % k;
    const uint kw = r % k;

    const uint p  = c / B;          // 位置优先列序 (p, b)
    const uint b  = c % B;
    const uint oh = p / OW;
    const uint ow = p % OW;

    const int ih = int(oh * stride + kh) - int(pad);
    const int iw = int(ow * stride + kw) - int(pad);

    float v = 0.0;
    if (ih >= 0 && ih < int(H) && iw >= 0 && iw < int(W))
        // 行主序：(row, b) → row * B + b
        v = in_data[(ci * H * W + uint(ih) * W + uint(iw)) * B + b];
    out_data[idx] = v;
}

// ── matmul_gemv.comp ──────────────────────────────────────────────────────
// 小 N 专用矩阵乘：C[M][N] = op(A) × op(B)，后端仅在 N ≤ 8 时分派本 kernel
// （借鉴 llama.cpp/ggml mul_mat_vec：线程沿 K 分片 + 寄存器累加 + subgroup
//   归约；tiled 版在小 N 下 64×64 块 56/64 列空转，本 kernel 每线程只算
//   真实的 N 列）。
//
// 结构：
//   - WorkGroup：256 线程（1D），覆盖 ROWS=4 个输出行 × N 列
//     （后端 dispatch grid.x = ceil(M/4)，两侧 ROWS 常量必须一致）
//   - 每线程持 k 步长 4 的分片（vec4 粒度），寄存器累加 acc[4][8]
//   - k 循环内零 barrier（各线程分片互不依赖），结束后一次归约写出
//   - 归约：subgroupAdd 每元素 → lane0 写 shared[num_subgroups] →
//     tid0 跨 subgroup 求和写出（同 gen_fused 归约 shader 先例；
//     shared 树形归约需 32KB partial/WG ≈ 每 WG 120KB 额外流量，
//     会吃掉 GEMV 的带宽收益，故不用）
//
// A 加载（按 trans 选连续维，vec4 对齐条件为 uniform 分支；未对齐走同
// 分组标量 + 边界判零，覆盖一致）：
//   transA=0：A[m][k] 行主序沿 k 连续 → K%4==0 时每行一条 vec4
//   transA=1：A 存储 (K,M) 沿 m 连续 → M%4==0 时每 k 一条 vec4（4 行同取）
// B 加载：
//   transB=0：块 [k0..k0+3][0..N) 扁平连续（k0%4 ⇒ k0*N%4==0 恒成立）
//     → k0+4≤K 即可 vec4；N%4==0 时按行 vec4 直取，否则扁平 vec4 拆分
//   transB=1：B 存储 (N,K) 沿 k 连续 → K%4==0 时每列一条 vec4
// 基址 16B 对齐由 MemoryPool 分配对齐 ≥16 保证（compute_memory_pool.hpp）
//
// push constants / descriptor 布局与 matmul.comp 完全一致（5×4B，3 binding），
// 复用 VulkanPipeline::create_matmul。
// ─────────────────────────────────────────────────────────────────────────

#version 450

#extension GL_EXT_control_flow_attributes : enable
#extension GL_KHR_shader_subgroup_basic : require
#extension GL_KHR_shader_subgroup_arithmetic : require

layout(local_size_x = 256, local_size_y = 1, local_size_z = 1) in;

layout(std430, binding = 0) readonly buffer  BufferA { float A[]; };
layout(std430, binding = 1) readonly buffer  BufferB { float B[]; };
layout(std430, binding = 2) writeonly buffer BufferC { float C[]; };

// vec4 别名视图（同 binding 双声明；仅在元素下标 %4==0 的快路径使用）
layout(std430, binding = 0) readonly buffer  BufferAv4 { vec4 Av4[]; };
layout(std430, binding = 1) readonly buffer  BufferBv4 { vec4 Bv4[]; };

layout(push_constant) uniform PushConstants {
    uint M;      // C 的行数
    uint N;      // C 的列数（后端保证 ≤ MAX_N）
    uint K;      // 求和维度
    uint transA; // 1 = A 存储为 (K,M)，按 A^T 使用
    uint transB; // 1 = B 存储为 (N,K)，按 B^T 使用
};

// 每 WG 覆盖的输出行数 —— 后端 dispatch（grid.x = ceil(M/ROWS)）必须同步
const uint ROWS = 4u;
// 支持的最大列数 —— 后端仅在 N ≤ MAX_N 时分派（acc/red 第二维）
const uint MAX_N = 8u;

// subgroup 部分和：256 线程 / subgroup ≥4 ⇒ 至多 64 个 subgroup
shared float red[ROWS][MAX_N][64];

void main()
{
    const uint tid = gl_LocalInvocationID.x;   // 0..255（沿 K 分片）
    const uint m0  = gl_WorkGroupID.x * ROWS;

    // ── 形状对齐标志（push 常量，uniform 分支零开销）─────────────────
    // 沿 k 连续：元素下标 m*K+k0%4==0 ∀m ⇔ K%4==0（k0 恒 %4）
    const bool a_vec_k = (transA == 0u) && ((K & 3u) == 0u);
    // 沿 m 连续（A 存 K×M）：(k*M+m0)%4==0 ∀k ⇔ M%4==0（m0=4*wg 恒 %4）
    // M%4==0 且 m0<M ⇒ m0+4≤M ⇒ 整组 4 行必在界内
    const bool a_vec_m = (transA != 0u) && ((M & 3u) == 0u);
    // B 存 (N,K) 沿 k 连续：n*K%4==0 ∀n ⇔ K%4==0（k0%4）
    const bool b_vec_k = (transB != 0u) && ((K & 3u) == 0u);

    // 行有效性（M%4≠0 时末 WG 行尾；M%4==0 时 m0<M ⇒ 全有效）
    bool row_ok[ROWS];
    [[unroll]] for (uint r = 0u; r < ROWS; ++r)
        row_ok[r] = (m0 + r < M);

    // ── 累加器（寄存器常驻；n≥N 恒 0，归约维度按 MAX_N 统一）────────
    float acc[ROWS][MAX_N];
    [[unroll]] for (uint r = 0u; r < ROWS; ++r)
        [[unroll]] for (uint n = 0u; n < MAX_N; ++n)
            acc[r][n] = 0.0;

    // 元素数上限（双重守卫：后端契约 N≤MAX_N，防越界 UB）
    const uint ncols = (N < MAX_N) ? N : MAX_N;

    // ── K 分片主循环：每线程步进 4*256=1024，分片互不依赖，零 barrier ──
    for (uint k0 = tid * 4u; k0 < K; k0 += 1024u)
    {
        // ── A 加载 → a[ROWS][4]（每线程持 4 个连续 k × ROWS 行）──────
        float a[ROWS][4];
        [[unroll]] for (uint r = 0u; r < ROWS; ++r)
            [[unroll]] for (uint i = 0u; i < 4u; ++i)
                a[r][i] = 0.0;

        if (transA == 0u)
        {
            if (a_vec_k)
            {
                // K%4==0 且 k0<K ⇒ k0+4≤K，整组 k 必在行内；行界单独判
                [[unroll]] for (uint r = 0u; r < ROWS; ++r)
                {
                    if (row_ok[r])
                    {
                        const vec4 v = Av4[((m0 + r) * K + k0) >> 2];
                        a[r][0] = v.x; a[r][1] = v.y;
                        a[r][2] = v.z; a[r][3] = v.w;
                    }
                }
            }
            else
            {
                [[unroll]] for (uint r = 0u; r < ROWS; ++r)
                {
                    if (!row_ok[r]) continue;
                    const uint base = (m0 + r) * K;
                    [[unroll]] for (uint i = 0u; i < 4u; ++i)
                        if (k0 + i < K) a[r][i] = A[base + k0 + i];
                }
            }
        }
        else
        {
            // A 存储 (K,M)：元素 (k,m) = A[k*M + m]
            if (a_vec_m)
            {
                // M%4==0 ⇒ 下标对齐且 m0+4≤M；k 逐个判界（K 可为奇数）
                [[unroll]] for (uint i = 0u; i < 4u; ++i)
                {
                    if (k0 + i < K)
                    {
                        const vec4 v = Av4[((k0 + i) * M + m0) >> 2];
                        [[unroll]] for (uint r = 0u; r < ROWS; ++r)
                            a[r][i] = v[r];
                    }
                }
            }
            else
            {
                [[unroll]] for (uint r = 0u; r < ROWS; ++r)
                {
                    if (!row_ok[r]) continue;
                    const uint m = m0 + r;
                    [[unroll]] for (uint i = 0u; i < 4u; ++i)
                        if (k0 + i < K) a[r][i] = A[(k0 + i) * M + m];
                }
            }
        }

        // ── B 加载 → b[4][MAX_N] ──────────────────────────────────────
        float b[4][MAX_N];
        [[unroll]] for (uint i = 0u; i < 4u; ++i)
            [[unroll]] for (uint n = 0u; n < MAX_N; ++n)
                b[i][n] = 0.0;

        if (transB == 0u)
        {
            // B 存储 (K,N)：元素 (k,n) = B[k*N + n]
            if (k0 + 4u <= K)
            {
                // 整组 k 在界内 ⇒ 扁平块 [k0*N, k0*N+4N) 连续且
                // k0*N%4==0（k0%4）⇒ 全部 vec4 对齐
                if ((N & 3u) == 0u)
                {
                    // N%4==0：按行 vec4 直取（(k0+i)*N%4==0 ⇔ N%4==0）
                    [[unroll]] for (uint i = 0u; i < 4u; ++i)
                    {
                        [[unroll]] for (uint j = 0u; j < 4u; ++j)
                        {
                            if (j * 4u >= ncols) break;
                            const vec4 v = Bv4[((k0 + i) * N + j * 4u) >> 2];
                            b[i][j * 4u + 0u] = v.x;
                            b[i][j * 4u + 1u] = v.y;
                            b[i][j * 4u + 2u] = v.z;
                            b[i][j * 4u + 3u] = v.w;
                        }
                    }
                }
                else
                {
                    // N%4≠0（1,2,3,5,6,7）：扁平 vec4 装入本地 cb，
                    // 按 p=i*N+n 拆分（N==1 时 cb[0] 即 4 个 k 的向量）
                    const uint flatn = (ncols * 4u + 3u) / 4u; // = ncols
                    vec4 cb[MAX_N];
                    [[unroll]] for (uint j = 0u; j < MAX_N; ++j)
                        cb[j] = vec4(0.0);
                    for (uint j = 0u; j < flatn; ++j)
                        cb[j] = Bv4[(k0 * N + j * 4u) >> 2];
                    [[unroll]] for (uint i = 0u; i < 4u; ++i)
                        [[unroll]] for (uint n = 0u; n < MAX_N; ++n)
                        {
                            if (n >= ncols) break;
                            const uint p = i * N + n;
                            b[i][n] = cb[p >> 2][p & 3u];
                        }
                }
            }
            else
            {
                // k 尾块（K%4≠0 的末段）：标量 + k 判界
                [[unroll]] for (uint i = 0u; i < 4u; ++i)
                {
                    if (k0 + i >= K) break;
                    [[unroll]] for (uint n = 0u; n < MAX_N; ++n)
                    {
                        if (n >= ncols) break;
                        b[i][n] = B[(k0 + i) * N + n];
                    }
                }
            }
        }
        else
        {
            // B 存储 (N,K)：元素 (k,n) = B[n*K + k]
            if (b_vec_k)
            {
                // K%4==0 且 k0<K ⇒ k0+4≤K；n<ncols≤N 恒在存储列界内
                [[unroll]] for (uint n = 0u; n < MAX_N; ++n)
                {
                    if (n >= ncols) break;
                    const vec4 v = Bv4[(n * K + k0) >> 2];
                    b[0][n] = v.x; b[1][n] = v.y;
                    b[2][n] = v.z; b[3][n] = v.w;
                }
            }
            else
            {
                [[unroll]] for (uint i = 0u; i < 4u; ++i)
                {
                    if (k0 + i >= K) break;
                    [[unroll]] for (uint n = 0u; n < MAX_N; ++n)
                    {
                        if (n >= ncols) break;
                        b[i][n] = B[n * K + k0 + i];
                    }
                }
            }
        }

        // ── 累加（外积 4×ncols，每元素 1 FMA）────────────────────────
        [[unroll]] for (uint r = 0u; r < ROWS; ++r)
        {
            if (!row_ok[r]) continue;
            [[unroll]] for (uint i = 0u; i < 4u; ++i)
            {
                const float av = a[r][i];
                [[unroll]] for (uint n = 0u; n < MAX_N; ++n)
                {
                    if (n >= ncols) break;
                    acc[r][n] = fma(av, b[i][n], acc[r][n]);
                }
            }
        }
    }

    // ── 归约：subgroupAdd 每元素（全 WG 同源收敛点，无发散）──────────
    [[unroll]] for (uint r = 0u; r < ROWS; ++r)
        [[unroll]] for (uint n = 0u; n < MAX_N; ++n)
        {
            acc[r][n] = subgroupAdd(acc[r][n]);
            if (gl_SubgroupInvocationID == 0u && gl_SubgroupID < 64u)
                red[r][n][gl_SubgroupID] = acc[r][n];
        }
    barrier();

    // ── 跨 subgroup 求和 + 写出（仅 tid0）────────────────────────────
    if (tid == 0u)
    {
        const uint nsub = (gl_NumSubgroups < 64u) ? gl_NumSubgroups : 64u;
        [[unroll]] for (uint r = 0u; r < ROWS; ++r)
        {
            if (m0 + r >= M) break;
            [[unroll]] for (uint n = 0u; n < MAX_N; ++n)
            {
                if (n >= ncols) break;
                float s = 0.0;
                for (uint g = 0u; g < nsub; ++g)
                    s += red[r][n][g];
                C[(m0 + r) * N + n] = s;
            }
        }
    }
}

// ── matmul_tiled.comp ─────────────────────────────────────────────────────
// vec4 向量化分块矩阵乘法：C[M][N] = A × B（借鉴 llama.cpp/ggml mul_mm 思路）
//
// 支持转置（同 matmul.comp）：
//   - transA=0, transB=0: C[M][N] = A[M][K] × B[K][N]
//   - transA=1, transB=0: C[M][N] = A^T[K,M] × B[K][N]
//   - transA=0, transB=1: C[M][N] = A[M][K] × B^T[N,K]
//   - transA=1, transB=1: C[M][N] = A^T[K,M] × B^T[N,K]
//
// 优化策略（对比旧标量 4×4 版：内层每 k 8 次标量 LDS + 16 FMA，2:1）：
//   - WorkGroup：16×16 线程，计算 64×64 输出块（每线程 4×4 元素）
//   - vec4 共享内存：Ash[k][m]（A 转置）/ Bsh[k][n]，每线程每 k 仅 2 次
//     vec4 LDS + 16 FMA（8:1 计算:加载比），消除旧版标量读的 bank 冲突
//   - vec4 累加器：acc[4]（4×4 = 16 个累加值），寄存器常驻
//
// v2（双缓冲流水，BK=16）：Ash/Bsh 各分 2 个 stage，tile t+1 的全局加载与
//   tile t 的计算并行发射；每 tile 的 barrier 由 2 次减为 1 次，全局内存
//   延迟（~600 周期）被计算覆盖。共享内存总计 16KB
//   （Ash[2][16][16]vec4 8KB + Bsh 8KB）——与单缓冲 BK=32 版相同的占用率；
//   BK=32 双缓冲需 32KB，会砍半 blocks/SM 抵消流水收益（实测净 0）。
//
// v3（vec4 全局加载/写出，借鉴 ggml mul_mm 的 ALIGNED/LOAD_VEC 快路径）：
//   - 每线程 EPT 个元素按「转置形态选连续维」分组，对齐时一条 vec4 全局
//     访问替代 4 次标量 + 逐元素边界判断；形状条件为 uniform 分支零开销
//   - 对齐条件（元素下标 %4==0 且组内连续 ⇔）：
//       A: transA=0 需 K%4==0（沿 k 连续）；transA=1 需 M%4==0（沿 m 连续）
//       B: transB=0 需 N%4==0（沿 n 连续）；transB=1 需 K%4==0（沿 k 连续）
//       C: 写出需 N%4==0（沿列连续）
//     基址 16B 对齐由 MemoryPool 分配对齐 ≥16 保证（compute_memory_pool.hpp）
//   - 未对齐（奇数形状）或线程持边界组：回退同组标量 + 边界判零，
//     覆盖分组与 vec4 路径完全一致 → 每个 (k,m) 恰被写一次，语义不变
//   - A/B 全局缓冲以 vec4 别名视图访问（同 binding 双声明，std430）
// ─────────────────────────────────────────────────────────────────────────

#version 450

// ── 精度变体（编译期开关：glslc -DNN_SHADER_F16=1）────────────────────────
// 默认（无宏）= 全 f32，**f32 分支文本逐字未改**（零回归）。
// 带宏 = f16 存储变体：缓冲声明 float16_t（GL_EXT_shader_16bit_storage），
//   别名槽改 uvec2（4×half = 8B，std430 步长恰好 8）→ unpackHalf2x16 解出
//   vec4；全局读处统一 float(...) 转 f32、写出处统一 float16_t(...)。
//   **共享 tile / vec4 外积累加 / 分块 / 双缓冲 / barrier 节奏与 f32 同构**，
//   只有全局加载/写出的元素类型变了（§7.2「f32 参考累加 + 输出舍入」）。
//   vec4 快路径的对齐条件不变：f16 下 uvec2 需 8B 对齐 ⇔ 元素下标 %4==0
//   （与 f32 的 vec4 16B 对齐条件同源，故 aligned 判据完全复用）。
#if defined(NN_SHADER_F16)
#extension GL_EXT_shader_16bit_storage : require
#define NN_ETYPE float16_t
#define NN_V4    uvec2
#define NN_LOADV4(arr, i) vec4(unpackHalf2x16(arr[i].x), unpackHalf2x16(arr[i].y))
#define NN_RD(x) float(x)
#define NN_WR(x) float16_t(x)
#else
#define NN_ETYPE float
#define NN_V4    vec4
#define NN_LOADV4(arr, i) arr[i]
#define NN_RD(x) x
#define NN_WR(x) x
#endif

layout(local_size_x = 16, local_size_y = 16) in;

layout(std430, binding = 0) readonly buffer  BufferA { NN_ETYPE A[]; };
layout(std430, binding = 1) readonly buffer  BufferB { NN_ETYPE B[]; };
layout(std430, binding = 2) writeonly buffer BufferC { NN_ETYPE C[]; };

// vec4 别名视图（同 binding 双声明；仅在元素下标 %4==0 的快路径使用）
layout(std430, binding = 0) readonly buffer  BufferAv4 { NN_V4 Av4[]; };
layout(std430, binding = 1) readonly buffer  BufferBv4 { NN_V4 Bv4[]; };
#if !defined(NN_SHADER_F16)
layout(std430, binding = 2) writeonly buffer BufferCv4 { vec4 Cv4[]; };
#endif

layout(push_constant) uniform PushConstants {
    uint M;      // C 的行数
    uint N;      // C 的列数
    uint K;      // 求和维度
    uint transA; // 1 = A 存储为 (K,M)，按 A^T 使用
    uint transB; // 1 = B 存储为 (N,K)，按 B^T 使用
};

// 分块维度常量
const uint BM = 64u;  // 输出块行数
const uint BN = 64u;  // 输出块列数
// BK=16 + 双缓冲（2026-09-24 OP/融合统一 A/B 定案）：Ash[2][16][16]vec4
//   8KB + Bsh 8KB = 16KB 共享。曾试与融合侧同参 BK=32 (32KB)：OP 级四点
//   **全回退**（512³ +9%、1024³ +26%、4096³ +29%、batched b8 +29%，
//   3 样本 best）——32KB 每 SM 只驻 2 个 WG，浅深网格都被占用率卡死；
//   OP 级恒取 16（下轮再统一须四点复测）。
const uint BK = 16u; // K 维度分块宽度

// 每线程协作加载元素数（BM*BK/256）。下方按 4 元素组取 groups=EPT/4 →
// 仅 BK%16==0 才铺满整 tile（BK=8 → groups=0 全不加载；BK=24 → 缺 1/3
// 垃圾值）——当前 BK=16 安全，改 BK 前先验 EPT%4==0。
const uint EPT_A = (BM * BK) / 256u;  // BK=16 → 4
const uint EPT_B = (BK * BN) / 256u;  // BK=16 → 4

// Shared Memory（vec4 布局，16B 对齐，双缓冲 [2][...]）：
//   Ash[stage][k][m]：A^T 分块 —— A[m][k]（转置写共享内存，内层按行连续读）
//   Bsh[stage][k][n]：B 分块 —— B[k][n]（内层按行连续读）
// 每个 vec4 = 4 个 float（64 行/列的 1/4）；stage 交替：tile t 用 t&1
shared vec4 Ash[2][BK][BM / 4];
shared vec4 Bsh[2][BK][BN / 4];

// ── 协作加载 A 分块 BM×BK → 指定 stage ────────────────────────────────────
// 线程分组：EPT_A/4 个「4 元素组」，组索引 g = tid*(EPT_A/4) + vv。
// 按 transA 选连续维（全局地址连续 ⇒ vec4 快路径）：
//   transA=0：A[m][k] 行主序，组 = 同行连续 4 个 k（g ↔ m=(4g)/BK, k0=(4g)%BK）
//   transA=1：A 存储 (K,M)，组 = 同 k 连续 4 个 m（g ↔ k=g/(BM/4), mq=g%(BM/4)）
// 未对齐/边界回退走**同一分组**的标量循环 → 覆盖恰好一致。
void load_A_tile(uint t, uint stage, uint tid, uint block_row)
{
    const uint a_base_col = t * BK;
    const bool k_contig = (transA == 0u);
    // vec4 元素下标 %4==0 的形状条件（uniform 分支）：
    //   沿 k 连续：gr*K%4==0 ∀gr ⇔ K%4==0；沿 m 连续：k*M%4==0 ∀k ⇔ M%4==0
    const bool aligned = k_contig ? ((K & 3u) == 0u) : ((M & 3u) == 0u);
    const uint groups = EPT_A / 4u;

    for (uint vv = 0u; vv < groups; ++vv)
    {
        const uint g = tid * groups + vv;
        if (k_contig)
        {
            // ── 沿 k 连续：4 个连续 k，同一行 m ──────────────────────────
            const uint m  = (4u * g) / BK;
            const uint k0 = (4u * g) % BK;
            const uint gr = block_row + m;
            const uint gc = a_base_col + k0;   // k0%4==0、a_base_col%4==0 ⇒ gc%4==0
            if (aligned && gr < M && gc < K)
            {
                // K%4==0 且 gc<K ⇒ gc+3<K，vec4 整组在行内；下标 %4==0 ⇒ 16B 对齐
                const vec4 v = NN_LOADV4(Av4, (gr * K + gc) >> 2);
                Ash[stage][k0    ][m >> 2][m & 3u] = v.x;
                Ash[stage][k0 + 1u][m >> 2][m & 3u] = v.y;
                Ash[stage][k0 + 2u][m >> 2][m & 3u] = v.z;
                Ash[stage][k0 + 3u][m >> 2][m & 3u] = v.w;
            }
            else
            {
                for (uint j = 0u; j < 4u; ++j)
                {
                    const float v = (gr < M && gc + j < K) ? NN_RD(A[gr * K + gc + j]) : 0.0;
                    Ash[stage][k0 + j][m >> 2][m & 3u] = v;
                }
            }
        }
        else
        {
            // ── 沿 m 连续（A 存储 K×M）：4 个连续 m，同一 k ─────────────
            const uint mq = g % (BM / 4u);
            const uint k  = g / (BM / 4u);
            const uint gc = a_base_col + k;
            const uint gr = block_row + mq * 4u;  // block_row%4==0（BM%4==0）
            if (aligned && gc < K && gr < M)
            {
                // M%4==0 且 gr<M ⇒ gr+3<M；下标 = gc*M+gr，两因子均 %4==0
                Ash[stage][k][mq] = NN_LOADV4(Av4, (gc * M + gr) >> 2);
            }
            else
            {
                for (uint j = 0u; j < 4u; ++j)
                {
                    const float v = (gr + j < M && gc < K) ? NN_RD(A[gc * M + gr + j]) : 0.0;
                    Ash[stage][k][mq][j] = v;
                }
            }
        }
    }
}

// ── 协作加载 B 分块 BK×BN → 指定 stage ────────────────────────────────────
// 按 transB 选连续维：
//   transB=0：B[k][n] 行主序，组 = 同 k 连续 4 个 n（g ↔ k=(4g)/BN, n0=(4g)%BN）
//   transB=1：B 存储 (N,K)，组 = 同 n 连续 4 个 k（g ↔ n=g/4, k0=(g%4)*4）
void load_B_tile(uint t, uint stage, uint tid, uint block_col)
{
    const uint b_base_row = t * BK;
    const bool n_contig = (transB == 0u);
    // 沿 n 连续：k*N%4==0 ∀k ⇔ N%4==0；沿 k 连续：n*K%4==0 ∀n ⇔ K%4==0
    const bool aligned = n_contig ? ((N & 3u) == 0u) : ((K & 3u) == 0u);
    const uint groups = EPT_B / 4u;

    for (uint vv = 0u; vv < groups; ++vv)
    {
        const uint g = tid * groups + vv;
        if (n_contig)
        {
            // ── 沿 n 连续：4 个连续 n，同一行 k ──────────────────────────
            const uint k  = (4u * g) / BN;
            const uint n0 = (4u * g) % BN;
            const uint gr = b_base_row + k;
            const uint gc = block_col + n0;     // n0%4==0、block_col%4==0 ⇒ gc%4==0
            if (aligned && gr < K && gc < N)
            {
                const vec4 v = NN_LOADV4(Bv4, (gr * N + gc) >> 2);
                Bsh[stage][k][n0 >> 2] = v;     // vec4 沿 n ↔ [k][n>>2] 整槽
            }
            else
            {
                for (uint j = 0u; j < 4u; ++j)
                {
                    const float v = (gr < K && gc + j < N) ? NN_RD(B[gr * N + gc + j]) : 0.0;
                    Bsh[stage][k][(n0 + j) >> 2][(n0 + j) & 3u] = v;
                }
            }
        }
        else
        {
            // ── 沿 k 连续（B 存储 N×K）：4 个连续 k，同一 n ─────────────
            const uint n  = g / 4u;
            const uint k0 = (g % 4u) * 4u;
            const uint gc = block_col + n;
            const uint gr = b_base_row + k0;    // b_base_row%4==0（t*BK%4==0）
            if (aligned && gc < N && gr < K)
            {
                const vec4 v = NN_LOADV4(Bv4, (gc * K + gr) >> 2);
                for (uint j = 0u; j < 4u; ++j)
                    Bsh[stage][k0 + j][n >> 2][n & 3u] = v[j];
            }
            else
            {
                for (uint j = 0u; j < 4u; ++j)
                {
                    const float v = (gr + j < K && gc < N) ? NN_RD(B[gc * K + gr + j]) : 0.0;
                    Bsh[stage][k0 + j][n >> 2][n & 3u] = v;
                }
            }
        }
    }
}

void main()
{
    const uint tx = gl_LocalInvocationID.x;   // 0..15
    const uint ty = gl_LocalInvocationID.y;   // 0..15
    const uint tid = ty * 16u + tx;           // 线性线程 ID 0..255

    // 当前 WorkGroup 负责的全局输出块起始坐标
    const uint block_row = gl_WorkGroupID.y * BM;
    const uint block_col = gl_WorkGroupID.x * BN;

    // ── 寄存器分块：4×4 累加器（vec4 数组） ─────────────────────────
    vec4 acc[4];
    for (uint i = 0u; i < 4u; ++i)
        acc[i] = vec4(0.0);

    const uint num_tiles = (K + BK - 1u) / BK;

    // ── 前奏：加载 tile 0 → stage 0，同步一次 ───────────────────────
    load_A_tile(0u, 0u, tid, block_row);
    load_B_tile(0u, 0u, tid, block_col);
    barrier();

    // ── 流水循环：发射 tile t+1 的加载（stage^1），同时计算 tile t ────
    //   每 tile 仅 1 次 barrier（旧版 2 次）；barrier 同时保证
    //   ① tile t+1 的全局加载在下一迭代被读取前完成
    //   ② tile t 的计算在 stage^1 被覆盖前完成（与 ①天然错开 stage）
    uint stage = 0u;
    for (uint t = 0u; t < num_tiles; ++t)
    {
        const uint nxt = t + 1u;
        const uint nstage = stage ^ 1u;
        if (nxt < num_tiles)
        {
            load_A_tile(nxt, nstage, tid, block_row);
            load_B_tile(nxt, nstage, tid, block_col);
        }

        // ── vec4 外积累加（每 k：2 次 vec4 LDS + 16 FMA = 8:1） ──────
        #pragma unroll 4
        for (uint k = 0u; k < BK; ++k)
        {
            // 本线程负责的 4 行（ty*4..ty*4+3）与 4 列（tx*4..tx*4+3）
            const vec4 a_vec = Ash[stage][k][ty];
            const vec4 b_vec = Bsh[stage][k][tx];

            acc[0] += a_vec.x * b_vec;
            acc[1] += a_vec.y * b_vec;
            acc[2] += a_vec.z * b_vec;
            acc[3] += a_vec.w * b_vec;
        }

        barrier();
        stage = nstage;
    }

    // ── 写入结果 ──────────────────────────────────────────────────────
#if defined(NN_SHADER_F16)
    // f16 输出：逐标量写出（C 无 vec4 别名视图），与 f32 的 N%4!=0 分支同构
    for (uint i = 0u; i < 4u; ++i)
    {
        const uint out_row = block_row + ty * 4u + i;
        if (out_row >= M) continue;
        const uint out_col0 = block_col + tx * 4u;
        if (out_col0 >= N) continue;
        C[out_row * N + out_col0 + 0u] = NN_WR(acc[i].x);
        if (out_col0 + 1u < N)
            C[out_row * N + out_col0 + 1u] = NN_WR(acc[i].y);
        if (out_col0 + 2u < N)
            C[out_row * N + out_col0 + 2u] = NN_WR(acc[i].z);
        if (out_col0 + 3u < N)
            C[out_row * N + out_col0 + 3u] = NN_WR(acc[i].w);
    }
#else
    // vec4 快路径（N%4==0，uniform）：out_col0%4==0 且 out_col0<N
    //   ⇒ out_col0+3<N（整组在行内），单条 vec4 写出免 3 次边界判断。
    // 否则逐标量 + 边界判断（列尾残余 / 奇数形状）。
    const bool c_vec4 = ((N & 3u) == 0u);
    for (uint i = 0u; i < 4u; ++i)
    {
        const uint out_row = block_row + ty * 4u + i;
        if (out_row >= M) continue;
        const uint out_col0 = block_col + tx * 4u;
        if (c_vec4)
        {
            if (out_col0 < N)
                Cv4[(out_row * N + out_col0) >> 2] = acc[i];
        }
        else
        {
            if (out_col0 >= N) continue;
            C[out_row * N + out_col0 + 0u] = acc[i].x;
            if (out_col0 + 1u < N)
                C[out_row * N + out_col0 + 1u] = acc[i].y;
            if (out_col0 + 2u < N)
                C[out_row * N + out_col0 + 2u] = acc[i].z;
            if (out_col0 + 3u < N)
                C[out_row * N + out_col0 + 3u] = acc[i].w;
        }
    }
#endif
}

// ── matmul.comp ─────────────────────────────────────────────────────────
// 分块矩阵乘法计算着色器：C[M][N] = A × B
//
// 支持转置：
//   - transA=0, transB=0: C[M][N] = A[M][K] × B[K][N]
//   - transA=1, transB=0: C[M][N] = A^T[K,M] × B[K][N]  （A 存储为 (K,M)）
//   - transA=0, transB=1: C[M][N] = A[M][K] × B^T[N,K]  （B 存储为 (N,K)）
//   - transA=1, transB=1: C[M][N] = A^T[K,M] × B^T[N,K]
//
// 工作原理：
//   - WorkGroup 大小：16 × 16（256 线程，适配大多数 GPU）
//   - 每个 WorkGroup 计算 16×16 的输出分块
//   - 共享内存（shared memory）缓存 A 和 B 的分块，减少全局内存访问
//   - 多轮迭代 K 维度，每轮加载 16×16 到 shared memory
// ─────────────────────────────────────────────────────────────────────────

#version 450

layout(local_size_x = 16, local_size_y = 16) in;

// Storage buffers（float32）
layout(std430, binding = 0) readonly buffer  BufferA { float A[]; };
layout(std430, binding = 1) readonly buffer  BufferB { float B[]; };
layout(std430, binding = 2) writeonly buffer BufferC { float C[]; };

// Push constants：矩阵维度 + 转置标志
layout(push_constant) uniform PushConstants {
    uint M;      // C 的行数
    uint N;      // C 的列数
    uint K;      // 求和维度
    uint transA; // 1 = A 存储为 (K,M)，按 A^T 使用
    uint transB; // 1 = B 存储为 (N,K)，按 B^T 使用
};

// 共享内存：缓存 A 和 B 的 16×16 分块
shared float sharedA[16][16];
shared float sharedB[16][16];

// ── 访问 A 元素 ─────────────────────────────────────────────────────────
// transA=0: A[m][k] = A[m * K + k]  （标准行主序 (M,K)）
// transA=1: A[m][k] = A[k * M + m]  （转置存储 (K,M)）
float getA(uint m, uint k) {
    return (transA != 0u) ? A[k * M + m] : A[m * K + k];
}

// ── 访问 B 元素 ─────────────────────────────────────────────────────────
// transB=0: B[k][n] = B[k * N + n]  （标准行主序 (K,N)）
// transB=1: B[k][n] = B[n * K + k]  （转置存储 (N,K)）
float getB(uint k, uint n) {
    return (transB != 0u) ? B[n * K + k] : B[k * N + n];
}

void main()
{
    const uint row = gl_GlobalInvocationID.y;  // 输出行
    const uint col = gl_GlobalInvocationID.x;  // 输出列
    const uint lid_x = gl_LocalInvocationID.x;
    const uint lid_y = gl_LocalInvocationID.y;

    float sum = 0.0;

    // K 维度的分块数
    const uint num_tiles = (K + 15u) / 16u;

    for (uint t = 0u; t < num_tiles; ++t)
    {
        // ── 从全局内存加载分块到共享内存 ──────────────────────────────
        const uint a_col = t * 16u + lid_x;
        if (row < M && a_col < K)
            sharedA[lid_y][lid_x] = getA(row, a_col);
        else
            sharedA[lid_y][lid_x] = 0.0;

        const uint b_row = t * 16u + lid_y;
        if (b_row < K && col < N)
            sharedB[lid_y][lid_x] = getB(b_row, col);
        else
            sharedB[lid_y][lid_x] = 0.0;

        barrier();

        // ── 累加部分积 ────────────────────────────────────────────────
        for (uint k = 0u; k < 16u; ++k)
            sum += sharedA[lid_y][k] * sharedB[k][lid_x];

        barrier();
    }

    // ── 写入结果 ──────────────────────────────────────────────────────
    if (row < M && col < N)
        C[row * N + col] = sum;
}

// ── outer_col.comp ────────────────────────────────────────────────────
// 逐列外积原语（backward 物化 dL/dA / dL/dB）：
//   out[(bh*dk+a)*dk+c, t] = P[a,t]·R[c,t] (· S[t] if has_scale)
// P/R (B*H*dk, seq); S (B*H*dk, seq) 标量头内逐行重复（读首行）；
// has_scale=0 时 S 为 dummy。
// 输出 (B*H*dk², seq)；扁平并行，无 shared memory。
// ─────────────────────────────────────────────────────────────────────
#version 450
layout(local_size_x = 256) in;

layout(std430, binding = 0) readonly  buffer BufP   { float P[]; };
layout(std430, binding = 1) readonly  buffer BufR   { float R[]; };
layout(std430, binding = 2) readonly  buffer BufS   { float S[]; };
layout(std430, binding = 3) writeonly buffer BufOut { float Out[]; };

layout(push_constant) uniform Push {
    uint dk, seq, rows, has_scale;
};

void main() {
    const uint idx   = gl_GlobalInvocationID.x;
    if (idx >= rows * dk * seq) return;

    const uint t  = idx % seq;
    const uint e  = idx / seq;
    const uint bh = e / (dk * dk);
    const uint rem = e % (dk * dk);
    const uint a  = rem / dk;
    const uint c  = rem % dk;
    const uint r0 = bh * dk;

    const float p = P[(r0 + a) * seq + t] * R[(r0 + c) * seq + t];
    Out[idx] = (has_scale == 1u) ? (p * S[r0 * seq + t]) : p;
}

// ── rearrange_3d.comp ───────────────────────────────────────────────────────
// 纯原语：3D 维度转置 (M, B, N) ↔ (B, M, N)
//
// 设计原则：本着色器只提供数据重排原语，不含任何算法语义。
//
// inverse=0: 输入 (M, B*N) → 输出 (B*M, N)
//   out[b*M + m, n] = in[m, b*N + n]
// inverse=1: 输入 (B*M, N) → 输出 (M, B*N)
//   out[m, b*N + n] = in[b*M + m, n]
//
// Push Constants：
//   uint M       — 第一维度大小
//   uint B       — 第二维度大小
//   uint N       — 第三维度大小
//   uint inverse — 0=forward, 1=inverse
//   uint total   — 元素总数 M*B*N
// ─────────────────────────────────────────────────────────────────────────

#version 450

layout(local_size_x = 256) in;

layout(std430, binding = 0) readonly buffer  BufferIn  { float in_data[];  };
layout(std430, binding = 1) writeonly buffer BufferOut { float out_data[]; };

layout(push_constant) uniform PushConstants {
    uint M;
    uint B;
    uint N;
    uint inverse;
    uint total;
};

void main()
{
    const uint idx = gl_GlobalInvocationID.x;
    if (idx >= total) return;

    // 把线性索引 idx 分解为 (m, b, n) 或 (b, m, n)
    if (inverse == 0u)
    {
        // 输入按 (M, B*N) 存储：idx = m * (B*N) + b * N + n
        const uint n = idx % N;
        const uint rest = idx / N;
        const uint b = rest % B;
        const uint m = rest / B;
        // 输出按 (B*M, N) 存储：out_idx = (b * M + m) * N + n
        const uint out_idx = (b * M + m) * N + n;
        out_data[out_idx] = in_data[idx];
    }
    else
    {
        // 输入按 (B*M, N) 存储：idx = (b * M + m) * N + n
        const uint n = idx % N;
        const uint rest = idx / N;
        const uint bm = rest;          // b * M + m
        const uint m = bm % M;
        const uint b = bm / M;
        // 输出按 (M, B*N) 存储：out_idx = m * (B * N) + b * N + n
        const uint out_idx = m * (B * N) + b * N + n;
        out_data[out_idx] = in_data[idx];
    }
}

// ── reduce.comp ───────────────────────────────────────────────────────────
// 归约原语着色器（工作组级并行归约）
//
// 只提供求和/最大值归约原语，供 Layer 组合表达 mean、var、grad_b 等。
//
// 模式：
//   mode=0: row_reduce — output[r] = reduce_op(A[r*cols + c]) over c
//           输入 (rows, cols) → 输出 (rows, 1)
//   mode=1: col_reduce — output[c] = reduce_op(A[r*cols + c]) over r
//           输入 (rows, cols) → 输出 (1, cols)
//
// 归约算子：
//   reduce_op=0: sum
//   reduce_op=1: max
//
// 绑定布局：
//   binding 0: 输入 A (readonly)
//   binding 1: 输出 OUT (writeonly)
//
// Push Constants：
//   uint rows      — 输入行数（两段式 pass2 = nchunk）
//   uint cols      — 输入列数
//   uint mode      — 0=row_reduce, 1=col_reduce
//   uint reduce_op — 0=sum, 1=max
//   uint chunk_rows — >0: 列归约两段式 pass1（WG.y = 行块序号，输出 = partials
//                    [y*cols + col]）；0 = 整表单段（输出 = result[col]）
//
// Dispatch（backend 侧，与下方各模式的 WG 结构契约同源）：
//   row_reduce: (rows, 1, 1)      — 每 WG 256 线程协作归约一行
//   col_reduce 单段: ((cols+31)/32, 1) — 每 WG 覆盖 32 列 tile
//   col_reduce 两段式（rows 大、单段 WG 数不足喂满机器时）：
//     pass1: ((cols+31)/32, nchunk, 1)，chunk_rows = ceil(rows/nchunk)
//     pass2: ((cols+31)/32, 1, 1)，输入 = partials(nchunk, cols)
//     两 pass 录入同一 cmd 单次提交（固定提交开销 ×1）——ggml
//     rms_norm_partials 的 partials-buffer 模式；动机：单段 WG 数 =
//     ceil(cols/32)（5244² 仅 164 vs 136 并发槽 → 1.2 波，末波只剩 ~20%
//     机器喂内存，实测带宽被尾部压约两成；对照 8192² 256 WG 尾部小带宽高）。
//
// ── 2026-09-24：与 DSL 生成归约（expr_glsl_gen generate_glsl_reduce）同步 ──
// 差异分解（此前两条路算法不一致，OP 级是慢的那条）：
//   ① 列归约访存：旧版「WG=单列、256 线程按行号跨步」——lane 间地址步进
//     cols×4B，完全不合并（每 warp 32 笔独立事务）；4096² 实测 159 GB/s
//     （同形状 row_reduce 482、add 412）。新版按列合并：lane = 列内偏移
//     （同一行 32 连续列 = 128B 单事务），行方向切成 8 个行块（warp id）
//     保并行度，跨行块部分和经 shared 合并（8 项，读写均合并、零 bank 冲突）。
//   ② max 恒等元：旧版 -inf（0xff800000）。全库统一 numeric_limits::lowest()
//     （0xFF7FFFFF，DSL/CPU 侧 2026-09 已改）——-inf 恒等元在全 -inf 输入
//     下与状态进位/减法组合会出 −inf−−inf=NaN（fold doc 掩码 GPU -nan 同类
//     根因）。
//   ③ 行归约归约树：旧版 shared 串行树（8 次 barrier）；同步为 DSL 的
//     subgroup 蝶式（3 次 barrier，subgroupAdd/subgroupMax → 跨 subgroup
//     收拢 → lane0 广播）。
//   4 路独立标量累加器（acc0..3，跨步 4*stride）与 DSL 行/列 pass 同款，
//   打破累加依赖链提升 ILP；归约结合序与单线程不同（测试容差覆盖，与
//   DSL 路径同等地位）。
// ─────────────────────────────────────────────────────────────────────────

#version 450

#extension GL_KHR_shader_subgroup_basic : require
#extension GL_KHR_shader_subgroup_arithmetic : require

layout(local_size_x = 256) in;

layout(std430, binding = 0) readonly buffer  BufIn  { float data[]; };
layout(std430, binding = 1) writeonly buffer BufOut { float result[]; };
// vec4 别名视图（同 binding 双声明；行归约 cols%4==0 快路径，16B 对齐由
// MemoryPool 保证）
layout(std430, binding = 0) readonly buffer  BufIn4 { vec4 data4[]; };

layout(push_constant) uniform PushConstants {
    uint rows;
    uint cols;
    uint mode;
    uint reduce_op;
    uint chunk_rows; // >0: 列归约两段式 pass1（WG.y = 行块序号）；0 = 整表单段
};

// 行模式：subgroup 蝶式归约暂存（256 线程 → ≤64 subgroup 部分和）
shared float sdata[256];
// 列模式：跨行块合并 [row_block 0..7][col_in_tile 0..31]
shared float sred[8][32];

// 恒等元：sum=0；max=lowest()（0xFF7FFFFF，全库统一，见文件头 ②）
float identity()
{
    return (reduce_op == 0u) ? 0.0 : uintBitsToFloat(0xFF7FFFFFu);
}

float combine(float a, float b)
{
    return (reduce_op == 0u) ? (a + b) : max(a, b);
}
vec4 combine(vec4 a, vec4 b)
{
    return (reduce_op == 0u) ? (a + b) : max(a, b);
}

void main()
{
    const uint tid = gl_LocalInvocationID.x;

    // ── 行归约：工作组 ID = 行索引，256 线程协作归约一行（合并访问）──
    if (mode == 0u)
    {
        const uint row = gl_WorkGroupID.x;
        if (row >= rows) return;

        const float init = identity();
        float v;

        if ((cols & 3u) == 0u)
        {
            // ── vec4 快路径（2026-09-24）：每线程 4 组 vec4 累加器，
            //    组内 4 元素连续 → 跨 lane 512B 连铺、指令数 ÷4。
            //    横折 4 分量后进 subgroup 蝶式（与标量回退同构）。
            const uint g4 = (row * cols) >> 2;
            const uint n4 = cols >> 2;
            vec4 acc0 = vec4(init), acc1 = vec4(init);
            vec4 acc2 = vec4(init), acc3 = vec4(init);
            uint g = tid;
            for (; g + 3u * 256u < n4; g += 4u * 256u)
            {
                acc0 = combine(acc0, data4[g4 + g]);
                acc1 = combine(acc1, data4[g4 + g + 256u]);
                acc2 = combine(acc2, data4[g4 + g + 512u]);
                acc3 = combine(acc3, data4[g4 + g + 768u]);
            }
            for (; g < n4; g += 256u)
                acc0 = combine(acc0, data4[g4 + g]);
            const vec4 s = combine(combine(acc0, acc1), combine(acc2, acc3));
            v = combine(combine(s.x, s.y), combine(s.z, s.w));
        }
        else
        {
            // 标量回退：4 路独立累加器（跨步 4*256），与 DSL 行 pass 同款 ILP
            float acc0 = init, acc1 = init, acc2 = init, acc3 = init;
            uint c = tid;
            for (; c + 3u * 256u < cols; c += 4u * 256u)
            {
                const uint base = row * cols;
                acc0 = combine(acc0, data[base + c]);
                acc1 = combine(acc1, data[base + c + 256u]);
                acc2 = combine(acc2, data[base + c + 512u]);
                acc3 = combine(acc3, data[base + c + 768u]);
            }
            for (; c < cols; c += 256u)
                acc0 = combine(acc0, data[row * cols + c]);
            v = combine(combine(acc0, acc1), combine(acc2, acc3));
        }

        sdata[tid] = v;
        barrier();

        // ── subgroup 蝶式归约（与 DSL emit_tree_reduce 同构）──────────
        v = sdata[tid];
        v = (reduce_op == 0u) ? subgroupAdd(v) : subgroupMax(v);
        if (gl_SubgroupInvocationID == 0u)
            sdata[gl_SubgroupID] = v;
        barrier();
        if (gl_SubgroupID == 0u)
        {
            float w = (gl_SubgroupInvocationID < gl_NumSubgroups)
                    ? sdata[gl_SubgroupInvocationID] : identity();
            w = (reduce_op == 0u) ? subgroupAdd(w) : subgroupMax(w);
            if (gl_SubgroupInvocationID == 0u)
                sdata[0] = w;
        }
        barrier();

        if (tid == 0u) result[row] = sdata[0];
        return;
    }

    // ── 列归约：WG = 32 列 tile；lane = 列内偏移（同 32 连续列 →
    //    同一行读地址连续 = 128B 合并事务），warp = 行块（8 个，r 步进 8）；
    //    跨行块部分和经 sred[8][32] 合并（写 wb*32+l 连续、读 w*32+l 连续）──
    if (mode == 1u)
    {
        const uint l  = tid & 31u;          // 列内偏移 0..31
        const uint wb = tid >> 5u;          // 行块 0..7
        const uint col0 = gl_WorkGroupID.x * 32u;
        const uint col = col0 + l;

        // 行窗口：chunk_rows>0（两段式 pass1）时 WG.y = 行块序号，本 WG 只
        // 归约 [rows_lo, rows_hi) 并写 partials[y*cols + col]；pass2 以
        // partials 为输入、rows=nchunk、chunk_rows=0 复用整表路径合并。
        const uint rows_lo = (chunk_rows == 0u) ? 0u : gl_WorkGroupID.y * chunk_rows;
        const uint rows_hi = (chunk_rows == 0u) ? rows
                                               : min(rows_lo + chunk_rows, rows);

        const float init = identity();
        float acc0 = init, acc1 = init, acc2 = init, acc3 = init;
        if (col < cols)
        {
            // 4 路累加器：跨步 4*8（与 DSL 列 pass 同款 ILP 形态）
            uint r = rows_lo + wb;
            for (; r + 3u * 8u < rows_hi; r += 4u * 8u)
            {
                acc0 = combine(acc0, data[r * cols + col]);
                acc1 = combine(acc1, data[(r + 8u) * cols + col]);
                acc2 = combine(acc2, data[(r + 16u) * cols + col]);
                acc3 = combine(acc3, data[(r + 24u) * cols + col]);
            }
            for (; r < rows_hi; r += 8u)
                acc0 = combine(acc0, data[r * cols + col]);
        }
        sred[wb][l] = combine(combine(acc0, acc1), combine(acc2, acc3));
        barrier();

        // 跨 8 行块合并（跨 lane 地址连续 = 合并读）
        float res = init;
        for (uint w = 0u; w < 8u; ++w)
            res = combine(res, sred[w][l]);

        if (wb == 0u && col < cols)
        {
            const uint out_idx = (chunk_rows == 0u) ? col
                                                   : gl_WorkGroupID.y * cols + col;
            result[out_idx] = res;
        }
        return;
    }
}

// ── scan_prefix_outer_gen.comp ───────────────────────────────────────────
// RLA 前缀扫描原语（**通用 d_k** 版本，dk 无 64 上限）：
//   当 d_k > 64 时后端走本 shader，A/B 运行态不再驻 shared（shared 是
//   O(dk²) 且受 48KB 预算限制），改驻**全局显存** scratch（B*H*2*dk*dk）。
//   共享内存只在 barrier 之间做进程同步，占用 O(1)，对任意 dk 安全。
// 语义与 scan_prefix_outer.comp 完全一致：
//   A_t = A0 + Σ_{i≤t,同文档} k_i·k_i^T,  B_t = B0 + Σ v_i·k_i^T
// 输出 (B*H*5*dk, seq) 行块：[0) B·P  [1) A·P  [2) B^T·R  [3) s  [4) r
// 1 workgroup/头，256 线程。性能弱于共享内存版本，优先保证正确性（任意 dk）。
// ─────────────────────────────────────────────────────────────────────────
#version 450
layout(local_size_x = 256) in;

layout(std430, binding = 0) readonly  buffer BufK   { float K[]; };
layout(std430, binding = 1) readonly  buffer BufV   { float V[]; };
layout(std430, binding = 2) readonly  buffer BufP   { float P[]; };
layout(std430, binding = 3) readonly  buffer BufR   { float R[]; };
layout(std430, binding = 4) readonly  buffer BufA0  { float A0[]; };
layout(std430, binding = 5) readonly  buffer BufB0  { float B0[]; };
layout(std430, binding = 6) readonly  buffer BufBnd { float Bnd[]; };
layout(std430, binding = 7)           buffer BufSt  { float St[]; };  // R/W scratch (B*H*2*dk*dk)
layout(std430, binding = 8) writeonly buffer BufOut { float Out[]; };

layout(push_constant) uniform Push {
    uint dk, heads, seq, causal, has_state, has_bnd, rows;
};

void main() {
    const uint bh  = gl_WorkGroupID.y;
    const uint b   = bh / heads;
    const uint h   = bh % heads;
    const uint r0  = bh * dk;
    const uint base = bh * 2u * dk * dk;   // A 在 [base, base+dk²)，B 紧随
    const uint tid  = gl_LocalInvocationID.x;

    // 1) 初始化 A/B（零 或 A0/B0 行块 h）
    for (uint idx = tid; idx < dk * dk; idx += 256u) {
        const uint a = idx / dk, c = idx % dk;
        float va = 0.0, vb = 0.0;
        if (has_state == 1u) {
            va = A0[(h * dk + a) * dk + c];
            vb = B0[(h * dk + a) * dk + c];
        }
        St[base + idx]           = va;
        St[base + dk * dk + idx] = vb;
    }
    barrier();

    // 2) 逐 t 顺序扫描 + 读出
    for (uint t = 0u; t < seq; ++t) {
        if (causal == 1u) {
            uint reset = 0u;
            if (has_bnd == 1u)
                reset = (Bnd[b * seq + t] != 0.0) ? 1u : 0u;
            for (uint idx = tid; idx < dk * dk; idx += 256u) {
                const uint a = idx / dk, c = idx % dk;
                const uint Aofs = base + idx, Bofs = base + dk * dk + idx;
                if (reset == 1u) { St[Aofs] = 0.0; St[Bofs] = 0.0; }
                St[Aofs] += K[(r0 + a) * seq + t] * K[(r0 + c) * seq + t];
                St[Bofs] += V[(r0 + a) * seq + t] * K[(r0 + c) * seq + t];
            }
        } else {
            for (uint idx = tid; idx < dk * dk; idx += 256u) {
                const uint a = idx / dk, c = idx % dk;
                const uint Aofs = base + idx, Bofs = base + dk * dk + idx;
                St[Aofs] += K[(r0 + a) * seq + t] * K[(r0 + c) * seq + t];
                St[Bofs] += V[(r0 + a) * seq + t] * K[(r0 + c) * seq + t];
            }
        }
        barrier();

        // num = B·q, Aq = A·q, B^T·R（第 tid 行，j/i 串行）
        if (tid < dk) {
            float num = 0.0, aq = 0.0;
            for (uint j = 0u; j < dk; ++j) {
                num += St[base + dk * dk + tid * dk + j] * P[(r0 + j) * seq + t];
                aq  += St[base + tid * dk + j]          * P[(r0 + j) * seq + t];
            }
            float btr = 0.0;
            for (uint i = 0u; i < dk; ++i)
                btr += St[base + dk * dk + i * dk + tid] * R[(r0 + i) * seq + t];
            Out[(r0 + tid) * seq + t]            = num;   // [0) B·P
            Out[(rows + r0 + tid) * seq + t]      = aq;   // [1) A·P
            Out[(2u * rows + r0 + tid) * seq + t] = btr;  // [2) B^T·R
        }
        barrier();

        // [3) s = q·(A·q)  [4) r = R·(B·q)：线程 0 全量归约
        if (tid == 0u) {
            float s = 0.0, r = 0.0;
            for (uint j = 0u; j < dk; ++j) {
                float aqj = 0.0, numj = 0.0;
                for (uint l = 0u; l < dk; ++l) {
                    aqj  += St[base + j * dk + l]          * P[(r0 + l) * seq + t];
                    numj += St[base + dk * dk + j * dk + l] * P[(r0 + l) * seq + t];
                }
                s += P[(r0 + j) * seq + t] * aqj;
                r += R[(r0 + j) * seq + t] * numj;
            }
            for (uint i = 0u; i < dk; ++i)
                Out[(3u * rows + r0 + i) * seq + t] = s;   // [3) 头内逐行重复
            for (uint i = 0u; i < dk; ++i)
                Out[(4u * rows + r0 + i) * seq + t] = r;   // [4) 头内逐行重复
        }
        barrier();
    }
}
// ── scan_prefix_outer.comp ─────────────────────────────────────────────
// RLA 前缀扫描原语：A_t = A0 + Σ_{i≤t, 同文档} k_i·k_i^T, B_t = B0 + Σ v_i·k_i^T
// causal=0: 全集常数（无重置）；causal=1: 含自身前缀，文档边界处运行态清零
// 输出 (B*H*5*dk, seq) 行块：[0) B·P  [1) A·P  [2) B^T·R  [3) s  [4) r
// 1 workgroup/头，256 线程；A/B 驻 shared 64×65×2≈33.3KB ≤ 850M 48KB
// ──────────────────────────────────────────────────────────────────────
#version 450
layout(local_size_x = 256) in;

layout(std430, binding = 0) readonly  buffer BufK  { float K[]; };
layout(std430, binding = 1) readonly  buffer BufV  { float V[]; };
layout(std430, binding = 2) readonly  buffer BufP  { float P[]; };
layout(std430, binding = 3) readonly  buffer BufR  { float R[]; };
layout(std430, binding = 4) readonly  buffer BufA0 { float A0[]; };
layout(std430, binding = 5) readonly  buffer BufB0 { float B0[]; };
layout(std430, binding = 6) readonly  buffer BufBnd{ float Bnd[]; };
layout(std430, binding = 7) writeonly buffer BufOut{ float Out[]; };

layout(push_constant) uniform Push {
    uint dk, heads, seq, causal, has_state, has_bnd, rows;
};

const uint PAD = 65u;
shared float sA[64][PAD];
shared float sB[64][PAD];
shared float s_kv[64], s_vv[64], s_qv[64], s_num[64], s_Aq[64];

void main() {
    const uint bh  = gl_WorkGroupID.y;
    const uint b   = bh / heads;
    const uint h   = bh % heads;
    const uint r0  = bh * dk;
    const uint tid = gl_LocalInvocationID.x;

    // 1) 初始化 A/B（零 或 A0/B0 行块 h）
    for (uint idx = tid; idx < dk * dk; idx += 256u) {
        const uint a = idx / dk, c = idx % dk;
        float va = 0.0, vb = 0.0;
        if (has_state == 1u) {
            const uint sr = h * dk + a;
            va = A0[sr * dk + c];
            vb = B0[sr * dk + c];
        }
        sA[a][c] = va;
        sB[a][c] = vb;
    }
    barrier();

    // 2) 双向：预累加全集
    if (causal == 0u) {
        for (uint t = 0u; t < seq; ++t) {
            if (tid < dk) {
                s_kv[tid] = K[(r0 + tid) * seq + t];
                s_vv[tid] = V[(r0 + tid) * seq + t];
            }
            barrier();
            for (uint idx = tid; idx < dk * dk; idx += 256u) {
                const uint a = idx / dk, c = idx % dk;
                sA[a][c] += s_kv[a] * s_kv[c];
                sB[a][c] += s_vv[a] * s_kv[c];
            }
            barrier();
        }
    }

    // 3) 逐 t 顺序扫描 + 读出
    for (uint t = 0u; t < seq; ++t) {
        if (tid < dk) {
            s_qv[tid] = P[(r0 + tid) * seq + t];
            s_kv[tid] = K[(r0 + tid) * seq + t];
            s_vv[tid] = V[(r0 + tid) * seq + t];
        }
        barrier();

        if (causal == 1u) {
            uint reset = 0u;
            if (has_bnd == 1u)
                reset = (Bnd[b * seq + t] != 0.0) ? 1u : 0u;
            for (uint idx = tid; idx < dk * dk; idx += 256u) {
                const uint a = idx / dk, c = idx % dk;
                if (reset == 1u) { sA[a][c] = 0.0; sB[a][c] = 0.0; }
                sA[a][c] += s_kv[a] * s_kv[c];
                sB[a][c] += s_vv[a] * s_kv[c];
            }
            barrier();
        }

        // num = B·qv; Aq = A·qv（dk 线程各处理一行，j 串行）
        if (tid < dk) {
            float acc = 0.0, acc2 = 0.0;
            for (uint j = 0u; j < dk; ++j) {
                acc  += sB[tid][j] * s_qv[j];
                acc2 += sA[tid][j] * s_qv[j];
            }
            s_num[tid] = acc;
            s_Aq[tid]  = acc2;
        }
        barrier();

        if (tid < dk) {
            float s = 0.0;
            for (uint j = 0u; j < dk; ++j)
                s += s_qv[j] * s_Aq[j];
            Out[(r0 + tid) * seq + t]              = s_num[tid]; // [0) B·P
            Out[(rows + r0 + tid) * seq + t]        = s_Aq[tid]; // [1) A·P
            Out[(3u * rows + r0 + tid) * seq + t]   = s;         // [3) s
        }
        // [2) B^T·R
        if (tid < dk) {
            float acc = 0.0;
            for (uint i = 0u; i < dk; ++i)
                acc += sB[i][tid] * R[(r0 + i) * seq + t];
            Out[(2u * rows + r0 + tid) * seq + t] = acc;
        }
        // [4) r = R·(B·P)，标量，线程 0 写全部行
        if (tid == 0u) {
            float r = 0.0;
            for (uint j = 0u; j < dk; ++j)
                r += R[(r0 + j) * seq + t] * s_num[j];
            for (uint i = 0u; i < dk; ++i)
                Out[(4u * rows + r0 + i) * seq + t] = r;
        }
        barrier();
    }
}

// ── scan_suffix_outer_gen.comp ───────────────────────────────────────────
// RLA 后缀扫描原语（**通用 d_k** 版本，dk 无 64 上限）：
//   d_k > 64 时后端走本 shader，S 运行态驻全局显存 scratch（B*H*dk*dk），
//   共享内存仅用于线程间同步、O(1)，对任意 dk 安全。
// 语义与 scan_suffix_outer.comp 完全一致：
//   causal=1: S_i = Σ_{t≥i,同文档} D_t（i+1 为文档起点时先清零）
//   causal=0: S_i = D_i
// D (B*H*dk², seq), X/Y (B*H*dk, seq)
// 输出 (B*H*3*dk, seq)：[0) S·X  [1) S·Y  [2) S^T·Y
// 1 workgroup/头，256 线程。性能弱于共享内存版本，优先保证正确性（任意 dk）。
// ─────────────────────────────────────────────────────────────────────────
#version 450
layout(local_size_x = 256) in;

layout(std430, binding = 0) readonly  buffer BufD   { float D[]; };
layout(std430, binding = 1) readonly  buffer BufX   { float X[]; };
layout(std430, binding = 2) readonly  buffer BufY   { float Y[]; };
layout(std430, binding = 3) readonly  buffer BufBnd { float Bnd[]; };
layout(std430, binding = 4)           buffer BufSt  { float St[]; };  // R/W scratch (B*H*dk*dk)
layout(std430, binding = 5) writeonly buffer BufOut { float Out[]; };

layout(push_constant) uniform Push {
    uint dk, heads, seq, causal, has_bnd, rows;
};

void main() {
    const uint bh    = gl_WorkGroupID.y;
    const uint b     = bh / heads;
    const uint r0    = bh * dk;
    const uint dbase = bh * dk * dk;
    const uint base  = bh * dk * dk;
    const uint tid   = gl_LocalInvocationID.x;

    // 初始化 S 为零
    for (uint idx = tid; idx < dk * dk; idx += 256u)
        St[base + idx] = 0.0;
    barrier();

    for (uint i = seq; i-- > 0u;) {
        if (causal == 1u) {
            uint reset = 0u;
            if (has_bnd == 1u && i + 1u < seq)
                reset = (Bnd[b * seq + i + 1u] != 0.0) ? 1u : 0u;
            for (uint idx = tid; idx < dk * dk; idx += 256u) {
                const uint a = idx / dk, c = idx % dk;
                if (reset == 1u) St[base + idx] = 0.0;
                St[base + idx] += D[(dbase + a * dk + c) * seq + i];
            }
        } else {
            for (uint idx = tid; idx < dk * dk; idx += 256u) {
                const uint a = idx / dk, c = idx % dk;
                St[base + idx] = D[(dbase + a * dk + c) * seq + i];
            }
        }
        barrier();

        if (tid < dk) {
            float sx = 0.0, sy = 0.0, syt = 0.0;
            for (uint c = 0u; c < dk; ++c) {
                sx  += St[base + tid * dk + c] * X[(r0 + c) * seq + i];
                sy  += St[base + tid * dk + c] * Y[(r0 + c) * seq + i];
            }
            for (uint r = 0u; r < dk; ++r)
                syt += St[base + r * dk + tid] * Y[(r0 + r) * seq + i];
            Out[(r0 + tid) * seq + i]               = sx;  // [0) S·X
            Out[(rows + r0 + tid) * seq + i]          = sy;  // [1) S·Y
            Out[(2u * rows + r0 + tid) * seq + i]     = syt; // [2) S^T·Y
        }
        barrier();
    }
}
// ── scan_suffix_outer.comp ────────────────────────────────────────────
// RLA 后缀扫描原语（backward pass 2）：
//   causal=1: S_i = Σ_{t≥i, 同文档} D_t（i+1 为文档起点时先清零）
//   causal=0: S_i = D_i（Layer 已把全集梯度沿 seq 广播）
// D (B*H*dk², seq), X/Y (B*H*dk, seq)
// 输出 (B*H*3*dk, seq)：[0) S·X  [1) S·Y  [2) S^T·Y
// 1 workgroup/头，256 线程；S 驻 shared 64×65≈16.6KB
// ─────────────────────────────────────────────────────────────────────
#version 450
layout(local_size_x = 256) in;

layout(std430, binding = 0) readonly  buffer BufD   { float D[]; };
layout(std430, binding = 1) readonly  buffer BufX   { float X[]; };
layout(std430, binding = 2) readonly  buffer BufY   { float Y[]; };
layout(std430, binding = 3) readonly  buffer BufBnd { float Bnd[]; };
layout(std430, binding = 4) writeonly buffer BufOut { float Out[]; };

layout(push_constant) uniform Push {
    uint dk, heads, seq, causal, has_bnd, rows;
};

const uint PAD = 65u;
shared float sS[64][PAD];
shared float s_xv[64], s_yv[64];

void main() {
    const uint bh   = gl_WorkGroupID.y;
    const uint b    = bh / heads;
    const uint r0   = bh * dk;
    const uint dbase = bh * dk * dk;
    const uint tid  = gl_LocalInvocationID.x;

    // 初始化 S 为零
    for (uint idx = tid; idx < dk * dk; idx += 256u)
        sS[idx / dk][idx % dk] = 0.0;
    barrier();

    for (uint i = seq; i-- > 0u;) {
        if (causal == 1u) {
            uint reset = 0u;
            if (has_bnd == 1u && i + 1u < seq)
                reset = (Bnd[b * seq + i + 1u] != 0.0) ? 1u : 0u;
            for (uint idx = tid; idx < dk * dk; idx += 256u) {
                const uint a = idx / dk, c = idx % dk;
                if (reset == 1u) sS[a][c] = 0.0;
                sS[a][c] += D[(dbase + a * dk + c) * seq + i];
            }
        } else {
            for (uint idx = tid; idx < dk * dk; idx += 256u) {
                const uint a = idx / dk, c = idx % dk;
                sS[a][c] = D[(dbase + a * dk + c) * seq + i];
            }
        }
        if (tid < dk) {
            s_xv[tid] = X[(r0 + tid) * seq + i];
            s_yv[tid] = Y[(r0 + tid) * seq + i];
        }
        barrier();

        if (tid < dk) {
            float sx = 0.0, sy = 0.0, syt = 0.0;
            for (uint c = 0u; c < dk; ++c) {
                sx += sS[tid][c] * s_xv[c];
                sy += sS[tid][c] * s_yv[c];
            }
            for (uint r = 0u; r < dk; ++r)
                syt += sS[r][tid] * s_yv[r];
            Out[(r0 + tid) * seq + i]               = sx;  // [0) S·X
            Out[(rows + r0 + tid) * seq + i]          = sy;  // [1) S·Y
            Out[(2u * rows + r0 + tid) * seq + i]     = syt; // [2) S^T·Y
        }
        barrier();
    }
}

// ── scatter_add.comp ──────────────────────────────────────────────────────
// 纯原语：按行索引原子累加梯度（scatter_add_rows）
//
// 语义：
//   dst: (vocab, D)    — 读写，原地累加
//   indices: (num,)    — 行索引（float 编码的整数）
//   grad: (num, D)     — 待累加梯度
//
//   dst[indices[i]][d] += grad[i][d]
//
// 使用 CAS 循环实现 float atomicAdd，兼容所有 Vulkan 设备。
// 越界索引被跳过（不修改 dst）。
//
// ⚠ 已知确定性偏差（见 docs/development/08-pitfalls-and-lessons.md 坑 19）：
//   CAS 循环的执行顺序取决于 GPU 调度，同一行多个梯度值的累加顺序
//   跨 run 不一致 → 浮点舍入结果有 ~1e-7 相对误差的 run 间抖动。
//   这是铁律 8（确定性）的**已文档化例外**：偏差仅为舍入噪声，被
//   优化器噪声吸收，不影响训练稳定性；CPU 参考路径（CpuEngine::
//   scatter_add_rows）保持顺序累加、逐字节确定。若未来需要 GPU 路径
//   也逐字节可复现，需改为"排序 + 顺序累加"或"分桶 + 确定性归约"。
//
// 绑定布局：
//   binding 0: dst (readwrite, coherent — float 数据以 uint 存储用于原子操作)
//   binding 1: indices (readonly)
//   binding 2: grad (readonly)
//
// Push Constants（12 bytes）：
//   uint vocab — dst 行数
//   uint D     — 列宽
//   uint num   — indices 元素个数
// ─────────────────────────────────────────────────────────────────────────

#version 450

layout(local_size_x = 256) in;

// dst 以 uint 声明，用于 atomicCompSwap 操作
// float 数据位模式直接存储为 uint（IEEE 754 位兼容）
layout(std430, binding = 0) coherent buffer BufDst     { uint  dst[];     };
layout(std430, binding = 1) readonly buffer BufIndices  { float indices[]; };
layout(std430, binding = 2) readonly buffer BufGrad     { float grad[];    };

layout(push_constant) uniform PushConstants {
    uint vocab;
    uint D;
    uint num;
};

// ── 基于 CAS 的 float atomicAdd ──────────────────────────────────────────
// 兼容所有 Vulkan 1.0 设备，无需 VK_EXT_shader_atomic_float 扩展。
void atomicAddFloat(uint idx, float val)
{
    uint old_v = dst[idx];
    uint new_v;
    while (true)
    {
        new_v = floatBitsToUint(uintBitsToFloat(old_v) + val);
        uint prev = atomicCompSwap(dst[idx], old_v, new_v);
        if (prev == old_v) break;
        old_v = prev;
    }
}

void main()
{
    const uint flat_idx = gl_GlobalInvocationID.x;
    const uint total = num * D;
    if (flat_idx >= total) return;

    const uint i = flat_idx / D;  // 第 i 个索引
    const uint d = flat_idx % D;  // 第 d 列

    const uint row_idx = uint(indices[i]);
    if (row_idx >= vocab) return;  // 越界跳过

    atomicAddFloat(row_idx * D + d, grad[flat_idx]);
}

// ── transpose.comp ────────────────────────────────────────────────────────
// 纯原语：矩阵转置 A (R, C) → OUT (C, R)
//
// out[c * R + r] = in[r * C + c]
//
// ── 64×64 分块 + 8×8 WG 砖块化 dispatch（2026-09-24 二轮定稿）─────────────
// 一轮（32×32 tile / 线性 dispatch）5244² 实测 0.958ms；二轮经三步消融定案：
//
//   消融链（5244²，wall，warmup 30 best-of-10）：
//     scale 纯拷贝（2N 连续流）                0.736
//     过 shared+barrier 恒等拷贝               0.754  ← shared/barrier 仅 ~2%
//     真转置 + 线性 dispatch                   0.942  ← +25% 来自写侧足迹散布
//     真转置 + 8×8 砖块化 dispatch             0.798  ← 本版本，−15%
//   结论：瓶颈不是 bank 冲突（全程零冲突）、不是段宽（64×64 vs 32×32 仅 2%）、
//   不是共享散布读写方向（落位交换 A/B 持平），而是**并发 WG 的访存足迹聚集度**：
//   转置输出行 = 输入列（block_c），线性 dispatch 下 136 个并发 WG 的写散布在
//   整块缓冲（5244² ≈ 180MB，每行仅写 256B、行距 21KB）→ DRAM/L2 写合并恶化。
//
//   砖块化：dispatch 网格 = (16→8, 8, n_bricks)，bz → 砖坐标，(x,y) → 砖内
//   tile。硬件按 x 最快线性调度 ⇒ 连续 64 个 WG 恰好填满一个 8×8 砖：
//     读侧每个砖行 = 64 输入行 × 8 段×256B 连铺；写侧每个砖列同构
//     → 两侧足迹 ~1MB 聚集（16×8 砖实测持平 8×8，取对称 8×8）。
//
//   其余定案（A/B 均已验证）：
//   - 64×64 tile（vs 32×32）：段宽 128B→256B、每线程 8→16 元素摊薄 barrier；
//   - 共享落位 tile[c_loc][r_loc]（阶段 2 连续读 + 连续写全局；阶段 1 散布写）：
//     两方向代价对称（A/B 持平），取阶段 2 全连续；
//   - bank 零冲突：行 padding 65，gcd(65,32)=1 ⇒ 列向跨 lane 步进命中 32 个
//     不同 bank；行向天然连续（阶段 1 散布写 addr=65x+常量 → bank=x+常量）；
//   - 无除/模（砖坐标一处 div/mod 每 WG 一次）；越界 tile 早退 + 元素守卫
//     读入/写出同构（r<rows && c<cols），未写共享槽必不被读。
//
// 共享内存：65 × 64 × 4B = 16.3KB。
//
// 绑定布局：
//   binding 0: 输入 A (readonly, R×C)
//   binding 1: 输出 OUT (writeonly, C×R)
//
// Push Constants（8 bytes）：
//   uint rows — 输入行数 R
//   uint cols — 输入列数 C
//
// Dispatch（backend 侧）：(8, 8, ceil(ceil(C/64)/8) * ceil(ceil(R/64)/8))
// ─────────────────────────────────────────────────────────────────────────

#version 450

layout(local_size_x = 32, local_size_y = 8) in;

layout(std430, binding = 0) readonly buffer  BufIn  { float in_data[];  };
layout(std430, binding = 1) writeonly buffer BufOut { float out_data[]; };

layout(push_constant) uniform PushConstants {
    uint rows;
    uint cols;
};

// 65 列 padding：gcd(65, 32) = 1 ⇒ 列向步进跨 32 lane 命中 32 个不同 bank
shared float tile[64][65];

void main()
{
    // ── 砖块化 dispatch：bz → 8×8 砖坐标，(x, y) → 砖内 tile 坐标 ──────
    const uint tx = (cols + 63u) / 64u;               // 列向 tile 数
    const uint bx_count = (tx + 7u) / 8u;             // 砖列数
    const uint bz = gl_WorkGroupID.z;
    const uint tile_c = (bz % bx_count) * 8u + gl_WorkGroupID.x;
    const uint tile_r = (bz / bx_count) * 8u + gl_WorkGroupID.y;
    if (tile_c * 64u >= cols || tile_r * 64u >= rows) return;

    const uint x  = gl_LocalInvocationID.x;   // 0..31 → C 方向（输入列/输出行）
    const uint y  = gl_LocalInvocationID.y;   // 0..7  → R 方向片段
    const uint block_c = tile_c * 64u;
    const uint block_r = tile_r * 64u;

    // ── 阶段 1：coalesced 读入（16 元素/线程）→ 散布写共享 ────────────
    for (uint a = 0u; a < 8u; ++a)
    {
        const uint r_loc = y + a * 8u;        // 覆盖行 0..63
        const uint r = block_r + r_loc;
        if (r >= rows) continue;
        for (uint h = 0u; h < 2u; ++h)
        {
            const uint c_loc = x + h * 32u;   // 覆盖列 0..63
            const uint c = block_c + c_loc;
            if (c < cols)
                tile[c_loc][r_loc] = in_data[r * cols + c];
        }
    }

    barrier();

    // ── 阶段 2：连续读共享 → coalesced 写全局（转置落位已在阶段 1 完成）
    // out[c*R + r]：跨 lane（x）r 连续 → 128B 合并，连续两次 = 256B 同行段
    for (uint b = 0u; b < 8u; ++b)
    {
        const uint c_loc = y + b * 8u;        // 覆盖列 0..63 → 输出行
        const uint c = block_c + c_loc;
        if (c >= cols) continue;
        for (uint g = 0u; g < 2u; ++g)
        {
            const uint r_loc = x + g * 32u;   // 覆盖行 0..63 → 输出列
            const uint r = block_r + r_loc;
            if (r < rows)
                out_data[c * rows + r] = tile[c_loc][r_loc];
        }
    }
}
